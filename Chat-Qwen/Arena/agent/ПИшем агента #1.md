COMPILED = [(re.compile(rx), risk, why) for rx, risk, why in RULES]

# символы, которые ломают «одна команда = одна программа»
SHELL_META = set(";|&$`><\n")


@dataclass
class Verdict:
    risk: Risk | None
    reason: str
    parsed: list[str] | None = None

    @property
    def allowed_automatically(self) -> bool:
        return self.risk is Risk.READ


def classify(command: str) -> Verdict:
    """Определить риск команды. Ничего не выполняет."""
    command = " ".join(command.strip().split())
    if not command:
        return Verdict(None, "пустая команда")

    if any(ch in SHELL_META for ch in command):
        return Verdict(None, "запрещены shell-операторы (; | && > $ `) — "
                             "передавайте одну команду без пайпов")

    for rx, risk, why in COMPILED:
        if rx.search(command):
            if risk is Risk.READ and not _read_paths_look_safe(command):
                return Verdict(None, "чтение файлов разрешено только из /var/log, /etc/sysagent "
                                     "и /proc")
            try:
                parsed = shlex.split(command)
            except ValueError as exc:
                return Verdict(None, f"не удалось разобрать команду: {exc}")
            return Verdict(risk, why, parsed)

    return Verdict(None, "команда не в белом списке политики (policy.py). "
                         "Добавьте правило осознанно, если она действительно нужна")


SAFE_READ_PREFIXES = ("/var/log", "/var/crash", "/proc", "/sys", "/etc/sysagent", "/tmp/sysagent")


def _read_paths_look_safe(command: str) -> bool:
    """Разрешаем читать файлы только из предсказуемых каталогов."""
    try:
        tokens = shlex.split(command)
    except ValueError:
        return False
    for token in tokens[1:]:
        if token.startswith("-"):
            continue
        if token.startswith("/") and not token.startswith(SAFE_READ_PREFIXES):
            return False
        if ".." in token:
            return False
    return True


class Auditor:
    """Журнал всех действий агента — отдельно от системного лога."""

    def __init__(self, path: Path):
        self.path = Path(path).expanduser()
        self.path.parent.mkdir(parents=True, exist_ok=True)

    def write(self, **fields) -> None:
        record = {
            "ts": time.strftime("%Y-%m-%dT%H:%M:%S%z"),
            "user": _safe_user(),
            **fields,
        }
        with self.path.open("a", encoding="utf-8") as fh:
            fh.write(json.dumps(record, ensure_ascii=False) + "\n")


def _safe_user() -> str:
    try:
        return getpass.getuser()
    except Exception:
        return "unknown"


def needs_confirmation(verdict: Verdict, mode: str, allow_dangerous: bool) -> bool:
    if verdict.risk is Risk.READ:
        return False
    if verdict.risk is Risk.DANGEROUS:
        return not allow_dangerous  # вызов вернёт отказ, см. guard()
    if mode == "readonly":
        return True  # будет отклонено
    return mode == "ask"


def guard(verdict: Verdict, mode: str, allow_dangerous: bool) -> tuple[bool, str]:
    """Финальное решение: можно ли выполнять без интерактивного подтверждения."""
    if verdict.risk is None:
        return False, f"запрещено: {verdict.reason}"
    if verdict.risk is Risk.DANGEROUS and not allow_dangerous:
        return False, f"запрещено (опасно: {verdict.reason})"
    if verdict.risk is Risk.WRITE and mode == "readonly":
        return False, "режим readonly: изменения запрещены"
    if verdict.risk is Risk.WRITE and mode == "ask":
        return False, "нужно подтверждение пользователя"
    return True, "разрешено"

@dataclass
class LLMConfig:
    base_url: str = "https://api.openai.com/v1"
    model: str = "gpt-4o-mini"
    api_key_env: str = "OPENAI_API_KEY"
    temperature: float = 0.1

    @property

@dataclass
class PolicyConfig:
    # readonly — только чтение (безопасный режим по умолчанию)
    # ask      — изменения требуют подтверждения в терминале
    # auto     — изменения выполняются без спроса (только для доверенных стендов!)
    mode: str = "ask"
    allow_dangerous: bool = False
    audit_log: Path = field(
        default_factory=lambda: Path("~/.local/state/sysagent/audit.jsonl").expanduser()
    )
    command_timeout: int = 20
    max_output_bytes: int = 20_000


@dataclass
class Config:
    llm: LLMConfig = field(default_factory=LLMConfig)
    policy: PolicyConfig = field(default_factory=PolicyConfig)
    source: Path | None = None

    @classmethod
    def load(cls, path: str | Path | None = None) -> "Config":
        cfg = cls()
        candidate: Path | None = Path(path).expanduser() if path else None
        if candidate is None:
            for p in DEFAULT_CONFIG_PATHS:
                if p.is_file():
                    candidate = p
                    break
        if candidate and candidate.is_file():
            data = tomllib.loads(candidate.read_text(encoding="utf-8"))
            cfg.source = candidate
            llm = data.get("llm", {})
            for key, value in llm.items():
                if hasattr(cfg.llm, key):
                    setattr(cfg.llm, key, value)
            pol = data.get("policy", {})
            for key, value in pol.items():
                if key == "audit_log":
                    value = Path(value).expanduser()
                if hasattr(cfg.policy, key):
                    setattr(cfg.policy, key, value)
        # переменные окружения имеют приоритет над файлом
        if os.environ.get("SYSAGENT_MODEL"):
            cfg.llm.model = os.environ["SYSAGENT_MODEL"]
        if os.environ.get("SYSAGENT_BASE_URL"):
            cfg.llm.base_url = os.environ["SYSAGENT_BASE_URL"]
        if os.environ.get("SYSAGENT_MODE") in {"readonly", "ask", "auto"}:
            cfg.policy.mode = os.environ["SYSAGENT_MODE"]
        return cfg

@dataclass
class ExecResult:

def run_command(command: str, cfg: Config, *, approve: Callable[[str, str], bool] | None = None,

    verdict = classify(command)
    risk = force_risk or verdict.risk

    if verdict.risk is None and force_risk is None:
        return ExecResult(command, False, -1, "", f"отклонено: {verdict.reason}", 0)

    ok_to_run, why = guard(verdict, cfg.policy.mode, cfg.policy.allow_dangerous)
    if not ok_to_run and approve is not None and verdict.risk is Risk.WRITE:
        ok_to_run = approve(command, verdict.reason)
        why = "подтверждено пользователем" if ok_to_run else "отклонено пользователем"

    if not ok_to_run:
        return ExecResult(command, False, -1, "", why, 0,
                          risk=risk.value if risk else None, denied_reason=why)

    try:
        proc = subprocess.run(  # noqa: S603 — аргументы разобраны shlex, shell=False
            verdict.parsed,
            capture_output=True,
            text=True,
            timeout=cfg.policy.command_timeout,
            env={**os.environ, "SYSTEMD_COLORS": "0", "LC_ALL": "C.UTF-8"},
        )
        stdout, stderr = _truncate(proc.stdout, cfg), _truncate(proc.stderr, cfg)
        res = ExecResult(command, proc.returncode == 0, proc.returncode, stdout, stderr,
                         int((time.monotonic() - started) * 1000),
                         risk=risk.value if risk else None)
    except FileNotFoundError:
        res = ExecResult(command, False, 127, "", "программа не найдена", 0,
                         risk=risk.value if risk else None)
    except subprocess.TimeoutExpired:
        res = ExecResult(command, False, -2, "", f"таймаут {cfg.policy.command_timeout}с", 0,
                         risk=risk.value if risk else None)

    Auditor(cfg.policy.audit_log).write(
        tool="run_command", command=command, risk=res.risk, ok=res.ok,
        exit_code=res.exit_code, denied=res.denied_reason,
    )
    return res


def _truncate(text: str, cfg: Config) -> str:
    limit = cfg.policy.max_output_bytes
    if len(text) <= limit:
        return text
    return text[:limit] + f"\n... [обрезано, всего {len(text)} байт]"

PRIORITY_NAMES = {"0": "emerg", "1": "alert", "2": "crit", "3": "err", "4": "warning",
                  "5": "notice", "6": "info", "7": "debug"}


def journal_errors(cfg: Config, unit: str | None = None, since: str = "-1h",
                   limit: int = 200, priority: str = "err",
                   approve=None) -> dict[str, Any]:
    """Ошибки из журнала systemd, сгруппированные по «шаблону» сообщения.

    Группировка важна: модели не нужно 200 одинаковых строк, ей нужны 5
    уникальных проблем и их количество.
    """
    if not shutil.which("journalctl"):
        return {"error": "journalctl не найден (не systemd-система?)"}
    parts = ["journalctl", "--no-pager", "-o", "json", "-p", priority]
    if unit:
        parts += ["-u", unit]
    parts += ["--since", since, "-n", str(max(1, min(limit, 2000)))]
    command = " ".join(parts)

    res = run_command(command, cfg, approve=approve, force_risk=Risk.READ)
    if res.denied_reason:
        return {"error": res.denied_reason}
    if res.exit_code != 0 and not res.stdout.strip():
        return {"error": res.stderr.strip() or f"journalctl вернул код {res.exit_code}"}

    events, parse_errors = [], 0
    for line in res.stdout.splitlines():
        line = line.strip()
        if not line:
            continue
        try:
            raw = json.loads(line)
        except json.JSONDecodeError:
            parse_errors += 1
            continue
        events.append({
            "ts": _ts(raw.get("__REALTIME_TIMESTAMP")),
            "unit": raw.get("_SYSTEMD_UNIT") or raw.get("SYSLOG_IDENTIFIER") or "?",
            "priority": PRIORITY_NAMES.get(str(raw.get("PRIORITY")), str(raw.get("PRIORITY"))),
            "pid": raw.get("_PID"),
            "message": _clean(raw.get("MESSAGE", "")),
        })

    return {
        "since": since,
        "unit": unit,
        "total_events": len(events),
        "groups": _cluster(events),
        "unparsed_lines": parse_errors,
    }


def _clean(message: Any) -> str:
    if isinstance(message, list):  # бинарные сообщения journald отдаёт списком байт
        try:
            message = bytes(message).decode("utf-8", "replace")
        except Exception:
            message = str(message)
    text = str(message)
    text = re.sub(r"\b0x[0-9a-fA-F]{6,}\b", "<addr>", text)
    text = re.sub(r"\b[0-9a-f]{8,}-[0-9a-f\-]{8,}\b", "<uuid>", text)
    text = re.sub(r"\b\d{2,}\b", "<n>", text)
    return text.strip()[:400]


def _cluster(events: list[dict]) -> list[dict]:
    buckets: dict[tuple, dict] = {}
    for ev in events:
        key = (ev["unit"], ev["message"])
        b = buckets.setdefault(key, {"unit": ev["unit"], "message": ev["message"],
                                     "count": 0, "first": ev["ts"], "last": ev["ts"],
                                     "priorities": set()})
        b["count"] += 1
        b["last"] = ev["ts"]
        b["priorities"].add(ev["priority"])
    out = []
    for b in buckets.values():
        b["priorities"] = sorted(b["priorities"])
        out.append(b)
    out.sort(key=lambda x: (-x["count"], x["unit"]))
    return out[:25]


def _ts(micro: Any) -> str:
    try:
        return time.strftime("%Y-%m-%d %H:%M:%S", time.localtime(int(micro) / 1_000_000))
    except Exception:
        return "?"

        return {"error": f"{type(exc).__name__}: {exc}"}

def handle_slash(line: str, cfg: Config) -> bool:
    """Возвращает False, если пора выходить."""
    parts = line.split(maxsplit=1)

    if cmd in {"/quit", "/exit", "/q"}:
        return False
    if cmd == "/help":
        print(BANNER)
    elif cmd == "/tools":

"""LLM-часть: цикл «модель -> инструмент -> наблюдение -> модель».

Используется любой OpenAI-совместимый эндпоинт (OpenAI, OpenRouter, Ollama,
vLLM, LM Studio), поэтому SDK не нужен — только stdlib urllib.

Замечание по безопасности: модель НЕ решает, можно ли выполнить команду.
Она лишь предлагает инструмент; решение принимает policy.py, а подтверждение
запрашивается у человека.
"""

from __future__ import annotations

import json
import urllib.error
import urllib.request
from typing import Any, Callable

from .config import Config
from .tools import TOOLS, call_tool, openai_tool_schemas

SYSTEM_PROMPT = """Ты — системный администратор Linux-хоста. Ты работаешь через инструменты,
которые дают доступ к журналу systemd, статусу служб и метрикам.

Правила:
1. Сначала диагностика, потом действие. Никогда не меняй систему, пока не понял причину.
2. Используй journal_errors с группировкой — не проси 1000 строк лога.
3. Перед изменением (restart/stop/install) кратко объясни: что делаешь, зачем, и какой риск.
4. Если инструмент вернул отказ политики — не пытайся обойти его другими командами.
   Сообщи пользователю, что нужно разрешить в policy.py.
5. Отвечай по-русски, кратко и структурно. Команды, которые предлагаешь выполнить
   вручную, оформляй в ```bash блоке.
6. Если данных не хватает — задай один точный вопрос, а не десять.
"""


class LLMError(RuntimeError):
    pass


def chat_completion(cfg: Config, messages: list[dict], tools: list[dict] | None = None,
                    timeout: int = 120) -> dict[str, Any]:
    api_key = cfg.llm.api_key
    if not api_key:
        raise LLMError(
            f"не задан ключ: export {cfg.llm.api_key_env}=... "
            "(для Ollama/LM Studio поставьте любое значение, напр. 'ollama')"
        )
    payload: dict[str, Any] = {
        "model": cfg.llm.model,
        "messages": messages,
        "temperature": cfg.llm.temperature,
    }
    if tools:
        payload["tools"] = tools
        payload["tool_choice"] = "auto"

    url = cfg.llm.base_url.rstrip("/") + "/chat/completions"
    request = urllib.request.Request(
        url,
        data=json.dumps(payload).encode(),
        headers={"Content-Type": "application/json",
                 "Authorization": f"Bearer {api_key}"},
        method="POST",
    )
    try:
        with urllib.request.urlopen(request, timeout=timeout) as response:  # noqa: S310
            return json.loads(response.read().decode())
    except urllib.error.HTTPError as exc:
        detail = exc.read().decode("utf-8", "replace")[:800]
        raise LLMError(f"HTTP {exc.code} от LLM: {detail}") from exc
    except urllib.error.URLError as exc:
        raise LLMError(f"не удалось соединиться с LLM ({url}): {exc.reason}") from exc


def run_agent_turn(cfg: Config, history: list[dict], user_message: str,
                   approve: Callable[[str, str], bool] | None = None,
                   verbose: bool = True) -> str:
    """Один ход диалога. history мутируется (дополняется) — для многоходового чата."""
    if not history:
        history.append({"role": "system", "content": SYSTEM_PROMPT})
    history.append({"role": "user", "content": user_message})

    schemas = openai_tool_schemas()
    for round_no in range(1, cfg.llm.max_tool_rounds + 1):
        data = chat_completion(cfg, history, tools=schemas)
        try:
            message = data["choices"][0]["message"]
        except (KeyError, IndexError) as exc:
            raise LLMError(f"неожиданный ответ LLM: {str(data)[:400]}") from exc

        tool_calls = message.get("tool_calls") or []
        if not tool_calls:
            answer = message.get("content") or ""
            history.append({"role": "assistant", "content": answer})
            return answer

        history.append({"role": "assistant", "content": message.get("content") or "",
                        "tool_calls": tool_calls})

        for call in tool_calls:
            fn = call["function"]["name"]
            try:
                args = json.loads(call["function"].get("arguments") or "{}")
            except json.JSONDecodeError:
                args = {}
            if verbose:
                print(f"  [tool {round_no}] {fn}({json.dumps(args, ensure_ascii=False)})")
            result = call_tool(fn, args, cfg, approve=approve)
            if verbose:
                preview = json.dumps(result, ensure_ascii=False)
                print(f"  [result] {preview[:300]}{'...' if len(preview) > 300 else ''}")
            history.append({"role": "tool", "tool_call_id": call.get("id", fn),
                            "name": fn,
                            "content": json.dumps(result, ensure_ascii=False)[:12_000]})

    return "Достигнут лимит раундов вызова инструментов — останавливаюсь. " \
           "Сузьте задачу или увеличьте llm.max_tool_rounds."

        params = ", ".join(spec["parameters"].get("properties", {}))
        flag = " [изменяет систему]" if spec.get("requires_approval") else ""
        lines.append(f"  {name}({params}){flag}\n      {spec['description'].splitlines()[0]}")
    return "\n".join(lines)

"""MCP-сервер (Model Context Protocol) поверх stdio.

Зачем: MCP — это «стандартный разъём», через который любой MCP-клиент
(Claude Desktop, IDE, свой чат) видит ваши инструменты как обычные tools.
Тогда ваш агент = MCP-сервер над системой, а UI/чат можно не писать вообще.

Запуск:  python3 -m sysagent.mcp_server --mode readonly
Подключение в claude_desktop_config.json:
  {"mcpServers": {"sysagent": {"command": "python3",
    "args": ["-m", "sysagent.mcp_server", "--mode", "readonly"]}}}
"""

from __future__ import annotations

import argparse
import json
import sys
from typing import Any

from .config import Config
from .tools import TOOLS, call_tool

PROTOCOL_VERSION = "2024-11-05"


def _tool_definitions() -> list[dict[str, Any]]:
    return [
        {"name": name,
         "description": spec["description"],
         "inputSchema": spec["parameters"] or {"type": "object", "properties": {}}}
        for name, spec in TOOLS.items()
    ]


def _handle(request: dict, cfg: Config) -> dict | None:
    method = request.get("method")
    req_id = request.get("id")
    params = request.get("params") or {}

    if method == "initialize":
        result = {
            "protocolVersion": PROTOCOL_VERSION,
            "capabilities": {"tools": {"listChanged": False}},
            "serverInfo": {"name": "sysagent", "version": "0.1.0"},
            "instructions": "Инструменты администрирования Linux. "
                            "Сначала диагностика (journal_errors, failed_units), потом действия.",
        }
    elif method == "tools/list":
        result = {"tools": _tool_definitions()}
    elif method == "tools/call":
        name = params.get("name", "")
        args = params.get("arguments") or {}
        # в MCP нет интерактивного подтверждения -> работает только policy (readonly/auto)
        payload = call_tool(name, args, cfg, approve=None)
        result = {"content": [{"type": "text", "text": json.dumps(payload, ensure_ascii=False)}],
                  "isError": "error" in payload}
    elif method == "ping":
        result = {}
    elif method is not None and method.startswith("notifications/"):
        return None  # уведомления не требуют ответа
    else:
        return {"jsonrpc": "2.0", "id": req_id,
                "error": {"code": -32601, "message": f"method not found: {method}"}}

    return {"jsonrpc": "2.0", "id": req_id, "result": result}


def main(argv: list[str] | None = None) -> int:
    parser = argparse.ArgumentParser(prog="sysagent-mcp")
    parser.add_argument("--config")
    parser.add_argument("--mode", default="readonly", choices=["readonly", "ask", "auto"])
    args = parser.parse_args(argv)

    cfg = Config.load(args.config)
    cfg.policy.mode = "readonly" if args.mode == "readonly" else args.mode

    for line in sys.stdin:
        line = line.strip()
        if not line:
            continue
        try:
            request = json.loads(line)
        except json.JSONDecodeError:
            print(json.dumps({"jsonrpc": "2.0", "id": None,
                              "error": {"code": -32700, "message": "parse error"}}),
                  flush=True)
            continue
        response = _handle(request, cfg)
        if response is not None:
            print(json.dumps(response, ensure_ascii=False), flush=True)
    return 0


if __name__ == "__main__":
    raise SystemExit(main())

"""Режим наблюдения: периодически смотрит журнал и сообщает о новых ошибках.

Это «дежурный» слой. Он принципиально read-only: ничего не перезапускает сам.
Автодействия без человека — источник инцидентов, включайте только после того,
как наберёте статистику и добавите чёткие правила (см. README, «Этап 4»).
"""

from __future__ import annotations

import argparse
import hashlib
import json
import subprocess
import sys
import time
from pathlib import Path

from .config import Config
from .tools import journal_errors

STATE = Path("~/.local/state/sysagent/watch-state.json").expanduser()


def load_state() -> set[str]:
    if STATE.is_file():
        try:
            return set(json.loads(STATE.read_text(encoding="utf-8"))["seen"])
        except (json.JSONDecodeError, KeyError):
            return set()
    return set()


def save_state(seen: set[str]) -> None:
    STATE.parent.mkdir(parents=True, exist_ok=True)
    STATE.write_text(json.dumps({"seen": sorted(seen)[-5000:]}), encoding="utf-8")


def notify(title: str, body: str) -> None:
    """Уведомление: в системный журнал (его увидят любые мониторинги) + stderr."""
    message = f"{title}: {body}"
    print(message, flush=True)
    if not sys.stdout.isatty():
        try:
            subprocess.run(["systemd-cat", "-t", "sysagent", "-p", "warning"],
                           input=message, text=True, timeout=5, check=False)  # noqa: S603
        except (FileNotFoundError, subprocess.SubprocessError):
            pass


def main(argv: list[str] | None = None) -> int:
    parser = argparse.ArgumentParser(prog="sysagent-watch")
    parser.add_argument("--config")
    parser.add_argument("--interval", type=int, default=60, help="период опроса, сек")
    parser.add_argument("--since", default="-2min", help="окно журнала за цикл")
    args = parser.parse_args(argv)

    cfg = Config.load(args.config)
    cfg.policy.mode = "readonly"  # дежурный контур никогда не меняет систему

    seen = load_state()
    print(f"sysagent-watch: интервал {args.interval}s, окно {args.since}, "
          f"audit={cfg.policy.audit_log}", flush=True)
    while True:
        try:
            report = journal_errors(cfg, since=args.since, limit=500)
            for group in report.get("groups", []):
                key = hashlib.sha256(
                    f"{group['unit']}|{group['message']}".encode()).hexdigest()[:16]
                if key in seen:
                    continue
                seen.add(key)
                notify("новая ошибка",
                       f"[{group['unit']}] x{group['count']} {group['message'][:200]}")
            save_state(seen)
        except KeyboardInterrupt:
            return 0
        except Exception as exc:  # дежурный цикл не должен падать
            print(f"watch error: {type(exc).__name__}: {exc}", flush=True)
        time.sleep(max(10, args.interval))


if __name__ == "__main__":
    raise SystemExit(main())

sysagent — локальный агент администрирования Linux
Рабочий скелет: читает ошибки журнала systemd, показывает состояние служб,
выполняет системные команды через политику безопасности, умеет работать как
обычный CLI-чат, как MCP-сервер и как дежурный демон.

Зависимостей нет — только Python 3.11+ (stdlib). Проверено на Debian 13 / systemd.

1. «Компилятор для агентов» — что это на самом деле
Компилятора агентов не существует, потому что агент — это обычная программа.
Никакой магии в ней нет: цикл «получить задачу → выбрать инструмент → выполнить →
посмотреть результат → снова выбрать». Компилятор/фреймворк не нужен, нужны три вещи:

Слой	Что делает	Чем делать
Инструменты (tools)	реальные действия: журнал, systemctl, файлы	subprocess, systemd D-Bus, psutil
Мозг	понимает запрос, выбирает инструменты	LLM с function calling или жёсткие правила
Политика	решает, что вообще можно выполнить	ваш код-белый список (обязательно!)
Так называемые «фреймворки для агентов» — это библиотеки-обвязки над этим циклом:

LangChain / LlamaIndex — много готового, но тяжёлые и быстро меняют API;
для системного агента это лишний слой абстракции.
Pydantic AI / smolagents / OpenAI Agents SDK — тоньше, но всё равно
привязывают вас к конкретному провайдеру моделей.
MCP (Model Context Protocol) — не фреймворк, а стандарт: вы описываете
инструменты один раз, и их видят Claude Desktop, IDE, собственный чат. Для
«агента над системой» это самый практичный выбор: UI писать не нужно.
Ansible / SaltStack — если задача на 80% про «привести сервер к состоянию»,
а не про диалог. Агент тогда только вызывает плейбуки.
Вывод: начинайте с 200–400 строк своего кода (как в этом репозитории). Когда
упрётесь в конкретную проблему — добавите библиотеку под неё.

2. Выбор языка
Python — быстрый прототип, богатые библиотеки для LLM, нормальная работа с
журналом и D-Bus. Рекомендуется для старта (этот скелет).
Go — один статический бинарник, без зависимостей на хосте, отличные
os/exec, log/syslog, D-Bus-клиент. Лучший выбор для продакшена на флот
серверов.
Rust — если нужны минимальный footprint и жёсткая безопасность памяти;
дольше в разработке.
Bash — только для триггеров, не для агента.

Принципиальные решения:

Модель не решает, что можно выполнять. LLM предлагает инструмент, а
policy.py решает: read → сразу, write → спросить человека, dangerous →
запрет по умолчанию, неизвестное → запрет.
shell=False всегда, аргументы разбираются shlex, пайпы и ; отклоняются
на этапе классификации (иначе journalctl | grep — это уже произвольный код).
Ошибки логов группируются по шаблону, а не отдаются модели пачкой строк:
50 × "upstream timed out <n>" вместо 50 одинаковых записей. Это резко
поднимает качество ответов модели и экономит токены.
Аудит: каждое действие пишется в ~/.local/state/sysagent/audit.jsonl.
Три режима: readonly (по умолчанию в дежурном контуре и в MCP),
ask (подтверждение в терминале), auto (только для тестовых стендов).

7. Безопасность — не пропускайте этот раздел
Агент с правом restart/kill — это удалённый root-доступ, если ошибиться в
одном месте. Минимальный набор:

Не выставляйте агент в интернет. Никаких 0.0.0.0:8080 без аутентификации.
Если нужен удалённый доступ — WireGuard/Tailscale/SSH-туннель, mTLS.
Белый список, а не чёрный. Запрещать rm бесполезно: find -delete,
> file, dd сделают то же. Разрешайте явные команды.
Человек в цикле для всего изменяющего. Первый месяц — обязательно.
Аудит-лог только на запись (chattr +a) и выгрузка наружу; иначе
скомпрометированный агент затрёт следы.
Ключ LLM — не root-овый. У агента свои права; LLM не должна видеть
секреты: вырезайте MESSAGE из журнала перед отправкой, если там бывают
токены/пароли.

// RunReadOnly запускает команду из фиксированного списка (без shell, без пайпов).
func RunReadOnly(name string, args ...string) (map[string]any, error) {
	if !readOnlyTools[name] {
		return nil, fmt.Errorf("программа %q не разрешена политикой", name)
	}
	for _, a := range args {
		if strings.ContainsAny(a, ";|&$`><\n") {
			return nil, fmt.Errorf("недопустимые символы в аргументе: %q", a)
		}
	}
	out, err := exec.Command(name, args...).CombinedOutput()
	code := 0
	if err != nil {
		if ee, ok := err.(*exec.ExitError); ok {
			code = ee.ExitCode()
		} else {
			return nil, err
		}
	}
	limit := 20000
	text := string(out)
	truncated := false
	if len(text) > limit {
		text, truncated = text[:limit], true
	}
	return map[string]any{
		"command": name + " " + strings.Join(args, " "),
		"output":  text, "exit_code": code, "truncated": truncated,
	}, nil
}

func JournalErrors(unit, since string, limit int, p *Policy) (any, error) {
	if _, err := exec.LookPath("journalctl"); err != nil {
		return nil, fmt.Errorf("journalctl не найден (не systemd-система?)")
	}
	if since == "" {
		since = "-1h"
	}
	if limit <= 0 || limit > 2000 {
		limit = 200
	}
	args := []string{"--no-pager", "-o", "json", "-p", "err", "--since", since, "-n", strconv.Itoa(limit)}
	if unit != "" {
		if !unitName.MatchString(unit) {
			return nil, fmt.Errorf("некорректное имя юнита")
		}
		args = append([]string{"-u", unit}, args...)
	}
	if p.Mode != "readonly" && p.Mode != "ask" && p.Mode != "auto" {
		return nil, fmt.Errorf("неизвестный режим политики %q", p.Mode)
	}

	res, err := RunReadOnly("journalctl", args...)
	if err != nil {
		return nil, err
	}
	output, _ := res["output"].(string)
	if code, _ := res["exit_code"].(int); code != 0 && strings.TrimSpace(output) == "" {
		return nil, fmt.Errorf("journalctl вернул код %d", code)
	}

	var groups []Group
	index := map[string]*Group{}
	total := 0
	for _, line := range strings.Split(output, "\n") {
		line = strings.TrimSpace(line)
		if line == "" {
			continue
		}
		var entry struct {
			Message  any    `json:"MESSAGE"`
			Unit     string `json:"_SYSTEMD_UNIT"`
			Ident    string `json:"SYSLOG_IDENTIFIER"`
			Realtime string `json:"__REALTIME_TIMESTAMP"`
		}
		if err := json.Unmarshal([]byte(line), &entry); err != nil {
			continue
		}
		total++
		u := entry.Unit
		if u == "" {
			u = entry.Ident
		}
		if u == "" {
			u = "?"
		}
		msg := normalizeMessage(fmt.Sprint(entry.Message))
		ts := formatMicro(entry.Realtime)
		key := u + "|" + msg
		if g, ok := index[key]; ok {
			g.Count++
			g.Last = ts
		} else {
			index[key] = &Group{Unit: u, Message: msg, Count: 1, First: ts, Last: ts}
		}
	}
	for _, g := range index {
		groups = append(groups, *g)
	}
	// сортировка: сначала самые частые
	for i := 0; i < len(groups); i++ {
		for j := i + 1; j < len(groups); j++ {
			if groups[j].Count > groups[i].Count {
				groups[i], groups[j] = groups[j], groups[i]
			}
		}
	}
	if len(groups) > 25 {
		groups = groups[:25]
	}
	return map[string]any{"since": since, "unit": unit, "total_events": total, "groups": groups}, nil
}

// runDaemon — «зависает в оперативке»: слушает unix-сокет и выполняет запросы.
// Ничего не делает в фоне, пока его не спросят (кроме режима watch).
func runDaemon(args []string) {
	fs := flag.NewFlagSet("daemon", flag.ExitOnError)
	socket := fs.String("socket", "", "путь к unix-сокету")
	mode := fs.String("mode", "ask", "readonly | ask | auto")
	_ = fs.Parse(args)

	path := socketPath(*socket)
	if err := ensureDir(path); err != nil {
		log.Fatalf("не создать каталог для сокета: %v", err)
	}
	_ = os.Remove(path) // сокет от прошлого запуска

	listener, err := net.Listen("unix", path)
	if err != nil {
		log.Fatalf("не удалось открыть сокет %s: %v", path, err)
	}
	// только владелец демона может им управлять
	if err := os.Chmod(path, 0o660); err != nil {
		log.Printf("предупреждение: chmod сокета: %v", err)
	}

	policy := &Policy{Mode: *mode, AllowDangerous: os.Getenv("SYSAGENTD_ALLOW_DANGEROUS") == "1"}

	log.Printf("sysagentd %s слушает %s (mode=%s, pid=%d)", Version, path, policy.Mode, os.Getpid())

	// корректное завершение: убрать сокет за собой
	stop := make(chan os.Signal, 1)
	signal.Notify(stop, syscall.SIGINT, syscall.SIGTERM)
	go func() {
		<-stop
		log.Println("получен сигнал, завершаюсь")
		_ = listener.Close()
		_ = os.Remove(path)
		os.Exit(0)
	}()

	for {
		conn, err := listener.Accept()
		if err != nil {
			return // listener закрыт -> штатный выход
		}
		go serveConn(conn, policy) // каждый запрос в своей горутине: стоит ~2 КБ
	}
}

func serveConn(conn net.Conn, policy *Policy) {
	defer conn.Close()
	started := time.Now()
	_ = conn.SetDeadline(time.Now().Add(90 * time.Second))

	var req Request
	if err := json.NewDecoder(conn).Decode(&req); err != nil {
		_ = json.NewEncoder(conn).Encode(Response{OK: false, Error: "плохой запрос: " + err.Error()})
		return
	}

	resp := handle(req, policy)
	resp.Elapsed = time.Since(started).Round(time.Millisecond).String()
	_ = json.NewEncoder(conn).Encode(resp)

	Audit(map[string]any{
		"cmd": req.Cmd, "args": req.Args, "ok": resp.OK,
		"error": resp.Error, "elapsed": resp.Elapsed,
	})
}

// handle — единственное место, где запрос превращается в действие.
// Здесь же (через policy.Check) решается, можно ли выполнить команду.
func handle(req Request, policies ...*Policy) Response {
	policy := &Policy{Mode: "readonly"}
	if len(policies) > 0 && policies[0] != nil {
		policy = policies[0]
	}
	data, err := dispatch(req, policy)
	if err != nil {
		return Response{OK: false, Error: err.Error()}
	}
	raw, err := json.Marshal(data)
	if err != nil {
		return Response{OK: false, Error: err.Error()}
	}
	return Response{OK: true, Data: raw}
}

func runClient(cmd string, args []string) {
	fs := flag.NewFlagSet(cmd, flag.ExitOnError)
	socket := fs.String("socket", "", "путь к unix-сокету демона")
	local := fs.Bool("local", false, "выполнить локально, без демона")
	unit := fs.String("unit", "", "имя systemd-юнита")
	since := fs.String("since", "-1h", "период журнала")
	limit := fs.Int("limit", 200, "сколько записей прочитать")
	_ = fs.Parse(args)

	req := Request{Cmd: cmd, Args: map[string]string{}}
	if *unit != "" {
		req.Args["unit"] = *unit
	}
	req.Args["since"] = *since
	req.Args["limit"] = fmt.Sprint(*limit)

	if cmd == "watch" {
		runWatch(*local, *socket, time.Duration(*limit)*time.Second)
		return
	}

	var resp Response
	if *local {
		resp = handle(req)
	} else {
		var err error
		resp, err = callDaemon(*socket, req)
		if err != nil {
			fmt.Fprintf(os.Stderr, "демон недоступен (%v)\n", err)
			fmt.Fprintln(os.Stderr, "запустите: sysagentd daemon   |   либо повторите с --local")
			os.Exit(1)
		}
	}

	if !resp.OK {
		fmt.Fprintln(os.Stderr, "ошибка:", resp.Error)
		os.Exit(1)
	}
	var pretty any
	_ = json.Unmarshal(resp.Data, &pretty)
	out, _ := json.MarshalIndent(pretty, "", "  ")
	fmt.Println(string(out))
}

func callDaemon(path string, req Request) (Response, error) {
	path = socketPath(path)
	conn, err := net.DialTimeout("unix", path, 3*time.Second)
	if err != nil {
		return Response{}, err
	}
	defer conn.Close()
	_ = conn.SetDeadline(time.Now().Add(60 * time.Second))

	if err := json.NewEncoder(conn).Encode(req); err != nil {
		return Response{}, err
	}
	var resp Response
	if err := json.NewDecoder(conn).Decode(&resp); err != nil {
		return Response{}, fmt.Errorf("не удалось прочитать ответ: %w", err)
	}
	return resp, nil
}

sysagentd — демон на Go: как он на самом деле работает
Мини-версия предыдущего Python-агента, но одним бинарником (3 МБ, без зависимостей).

Ответ на вопрос «запустил — висит в памяти — начинаю работать»
Да, но с важным уточнением: демон сам по себе не даёт вам никакого набора команд.
Интерфейс вы проектируете сами. «Свой набор команд» появляется только потому, что
в том же бинарнике есть клиентская часть, которая разговаривает с демоном.
Тут их две в одном файле:

text

$ sysagentd daemon            # роль 1: слушает unix-сокет, висит в памяти

$ sysagentd logs --local      # роль 2б: тот же запрос без демона (для cron)
Важно: sysagentd logs — это не «команда внутри демона» и не отдельная
утилита. Это тот же бинарник, запущенный второй раз; он открывает unix-сокет
/run/sysagentd.sock, отправляет JSON, получает JSON, печатает, выходит.
Демон при этом продолжает висеть и помнит состояние (uptime, кэш, счётчики).

Способ	Как выглядит	Когда использовать

Тут важно различать три вещи:

Демон — исполнитель: проверяет запрос по политике, запускает разрешённый инструмент и возвращает результат.

Про Claude и SSH
Есть важное уточнение: облачная модель Claude сама по себе не «зайдёт по SSH» на ваш компьютер или сервер. Между ней и Linux должен быть посредник — например, клиент или адаптер на вашем ПК, который получает запросы Claude и передаёт их демону по SSH.

Если речь о локальном приложении Claude на ПК, можно будет рассмотреть MCP или такой адаптер. Если вы имеете в виду сайт Claude.ai в браузере, понадобится поддерживаемый способ подключения или отдельный мост. Конкретный вариант зависит от того, каким именно клиентом Claude вы пользуетесь.

При этом не нужно давать Claude полный SSH-доступ или root-shell. Лучше предоставить ему только узкие операции через адаптер. И если данные из журналов отправляются облачной модели, нужно заранее решить, какие именно строки ей можно передавать и как маскировать секреты.

С чего разумно начать
Я бы начал с одного Linux-хоста и только read-only:

Демон работает как отдельная служба systemd и принимает запросы через локальный Unix-сокет — без открытого сетевого порта.
CLI-клиент умеет запросить журнал, статус служб, ошибки и состояние диска.
Демон сам проверяет разрешённость каждой операции и ведёт аудит.
Когда локальный вариант надёжен, подключается удалённый клиент через ограниченный SSH-канал.

Коротко по вашему вопросу
«Компилятора для агентов» не существует, потому что агент — это обычная программа. В ней всего три слоя:

Инструменты (tools) — реальные действия: чтение журнала, systemctl, файлы.
Мозг — LLM с function calling или жёсткие правила; опционален.
Политика — ваш код, который решает, что вообще разрешено выполнять. Это самый важный слой, и его нельзя отдавать модели.
Из готового: LangChain/LlamaIndex — тяжёлые обвязки, для системного агента скорее мешают; Pydantic AI / OpenAI Agents SDK — тоньше, но привязывают к провайдеру модели; MCP (Model Context Protocol) — это не фреймворк, а стандарт, и для «агента над системой» самый практичный: описываете инструменты один раз, и их видит любой MCP-клиент, UI писать не надо. Ansible — если задача про «привести сервер к состоянию», а не про диалог.

Язык: Python для старта (быстрее всего), Go — если нужен один бинарник без зависимостей на флоте серверов.

Что уже работает в скелете (~1000 строк, без внешних зависимостей)
journal_errors — ошибки из журнала systemd, сгруппированные по шаблону с подсчётом повторов (50×upstream timed out <n> вместо 50 одинаковых строк — это резко улучшает ответы модели).
failed_units, unit_status, disk_usage, memory_usage, top_processes, tail_log_file, run_shell_command.
Политика с классификацией read / write / dangerous: чтение идёт сразу, systemctl restart спрашивает подтверждение, rm/reboot/useradd/iptables/curl/sudo запрещены по умолчанию, неизвестная команда — запрещена. shell=False, пайпы и ; отклоняются. Всё пишется в аудит-лог.
Три интерфейса: CLI-чат со слэш-командами (/failed, /logs nginx.service — работают даже без ключа LLM), MCP-сервер по stdio, и дежурный демон watch.py с systemd-юнитом (уже с изоляцией: ProtectSystem=strict, NoNewPrivileges, пустой CapabilityBoundingSet).
Поддержка любого OpenAI-совместимого API, включая локальную Ollama (можно вообще без облака).
Проверено: python3 tests/test_policy.py проходит; systemctl restart nginx без подтверждения не выполняется, sudo rm -rf / отклоняется даже в режиме auto; MCP-сервер корректно отвечает на initialize / tools/list / tools/call.
