Почему это лучший вариант на практике
Почти все современные площадки с поддержкой вызова инструментов (tools) работают по одному и тому же протоколу — OpenAI-совместимому JSON API (/chat/completions). Значит, мы пишем один компактный клиент на стандартном Go, а в конфиг агента (/etc/sysagentd/llm.jsonc с правами 0600) вы добавляете сразу 2–4 бесплатных профиля.

Как это решает все три ваши задачи:

Бесплатно: используются только бесплатные тарифы без привязки банковской карты.
Обход лимитов автоматом (Failover): если модель №1 упёрлась в минутный или дневной лимит (HTTP 429) или задумалась по таймауту, sysagentd сам прозрачно переключает запрос на модель №2 → №3, а уставшую модель ставит на паузу «остывания» (например, на 60 секунд). Лимиты всех подключённых площадок по сути складываются.
Вариации на выбор: прямо в меню sysagent [●]> добавим пункт переключения модели: можно держать режим auto (каскад), а можно одной кнопкой выбрать конкретную модель (например, сравнить, что скажет GPT-4.1, Qwen 3, Llama 3.3 или Mistral по одной и той же проблеме). Плюс туда же без переделки кода в любой момент подключается локальная Ollama внутри вашей сети.

func normalizeUnit(raw string) (string, error) {
	u := strings.TrimSpace(raw)
	if u == "" || strings.HasPrefix(u, "-") || !validUnitRe.MatchString(u) {
		return "", fmt.Errorf("недопустимое имя службы %q", raw)
	}
	if !strings.Contains(u, ".") {
		u += ".service"
	}
	return u, nil
}

func runSummary(_ context.Context, _ map[string]string) (any, error) {
	host, _ := os.Hostname()
	osName := readOSRelease()
	kernel := readTrimmedFile("/proc/sys/kernel/osrelease")
	uptimeSec, uptimeHuman := readUptime()
	load1, load5, load15 := readLoadAvg()
	mem := readMemInfo()
	disks := readDisks([]string{"/", "/var"})

	return map[string]any{
		"hostname":     host,
		"os":           osName,
		"kernel":       kernel,
		"cpu_cores":    runtime.NumCPU(),
		"uptime_sec":   uptimeSec,
		"uptime_human": uptimeHuman,
		"load_avg": map[string]float64{
			"1m":  load1,
			"5m":  load5,
			"15m": load15,
		},
		"memory_mb": mem,
		"disks":     disks,
	}, nil
}

func readOSRelease() string {
	f, err := os.Open("/etc/os-release")
	if err != nil {
		return runtime.GOOS
	}
	defer f.Close()
	sc := bufio.NewScanner(f)
	for sc.Scan() {
		line := sc.Text()
		if strings.HasPrefix(line, "PRETTY_NAME=") {
			return strings.Trim(strings.TrimPrefix(line, "PRETTY_NAME="), `"'`)
		}
	}
	return runtime.GOOS
}

func readTrimmedFile(path string) string {
	b, err := os.ReadFile(path)
	if err != nil {
		return "unknown"
	}
	return strings.TrimSpace(string(b))
}

func readUptime() (int64, string) {
	raw := readTrimmedFile("/proc/uptime")
	fields := strings.Fields(raw)
	if len(fields) == 0 {
		return 0, "unknown"
	}
	secFloat, err := strconv.ParseFloat(fields[0], 64)
	if err != nil {
		return 0, "unknown"
	}
	sec := int64(secFloat)
	return sec, (time.Duration(sec) * time.Second).String()
}

func readLoadAvg() (float64, float64, float64) {
	fields := strings.Fields(readTrimmedFile("/proc/loadavg"))
	if len(fields) < 3 {
		return 0, 0, 0
	}
	l1, _ := strconv.ParseFloat(fields[0], 64)
	l5, _ := strconv.ParseFloat(fields[1], 64)
	l15, _ := strconv.ParseFloat(fields[2], 64)
	return l1, l5, l15
}

func readMemInfo() map[string]any {
	vals := make(map[string]uint64)
	f, err := os.Open("/proc/meminfo")
	if err == nil {
		defer f.Close()
		sc := bufio.NewScanner(f)
		for sc.Scan() {
			parts := strings.Fields(sc.Text())
			if len(parts) >= 2 {
				key := strings.TrimSuffix(parts[0], ":")
				v, err := strconv.ParseUint(parts[1], 10, 64)
				if err == nil {
					vals[key] = v
				}
			}
		}
	}
	totalMB := vals["MemTotal"] / 1024
	availMB := vals["MemAvailable"] / 1024
	var usedMB uint64
	if totalMB > availMB {
		usedMB = totalMB - availMB
	}
	var usedPct float64
	if totalMB > 0 {
		usedPct = float64(usedMB*1000/totalMB) / 10.0
	}
	return map[string]any{
		"total_mb":     totalMB,
		"used_mb":      usedMB,
		"available_mb": availMB,
		"used_percent": usedPct,
		"swap_total":   vals["SwapTotal"] / 1024,
		"swap_free":    vals["SwapFree"] / 1024,
	}
}

func readDisks(paths []string) []map[string]any {
	seenDev := make(map[uint64]bool)
	var out []map[string]any
	for _, p := range paths {
		fi, err := os.Stat(p)
		if err != nil {
			continue
		}
		if st, ok := fi.Sys().(*syscall.Stat_t); ok {
			if seenDev[uint64(st.Dev)] {
				continue
			}
			seenDev[uint64(st.Dev)] = true
		}
		var fs syscall.Statfs_t
		if err := syscall.Statfs(p, &fs); err != nil || fs.Blocks == 0 {
			continue
		}
		totalBytes := fs.Blocks * uint64(fs.Bsize)
		availBytes := fs.Bavail * uint64(fs.Bsize)
		usedBytes := totalBytes - (fs.Bfree * uint64(fs.Bsize))
		pct := float64(usedBytes*1000/totalBytes) / 10.0
		out = append(out, map[string]any{
			"mount":        p,
			"total_gb":     float64(totalBytes/(1024*1024*102)) / 10.0,
			"used_gb":      float64(usedBytes/(1024*1024*102)) / 10.0,
			"available_gb": float64(availBytes/(1024*1024*102)) / 10.0,
			"used_percent": pct,
		})
	}
	return out
}

func runLogs(ctx context.Context, args map[string]string) (any, error) {
	since := strings.TrimSpace(args["since"])
	if since == "" {
		since = "-1h"
	}
	if !validSpanRe.MatchString(since) {
		return nil, fmt.Errorf("параметр since должен иметь вид -15m, -1h, -24h (получено %q)", since)
	}

	cmdArgs := []string{"-p", "err..alert", "--no-pager", "-n", "200", "--since", since}
	if rawUnit := strings.TrimSpace(args["unit"]); rawUnit != "" {
		u, err := normalizeUnit(rawUnit)
		if err != nil {
			return nil, err
		}
		cmdArgs = append(cmdArgs, "-u", u)
	}

	out, err := exec.CommandContext(ctx, "journalctl", cmdArgs...).CombinedOutput()
	if err != nil && len(out) == 0 {
		return nil, fmt.Errorf("ошибка чтения journalctl: %w", err)
	}

	type grouped struct {
		Message string `json:"message"`
		Count   int    `json:"count"`
	}
	counts := make(map[string]int)
	var order []string
	totalRaw := 0

	for _, rawLine := range strings.Split(string(out), "\n") {
		line := strings.TrimSpace(rawLine)
		if line == "" || strings.HasPrefix(line, "-- ") {
			continue
		}
		totalRaw++
		normalized := scrubLogLine(stripJournalPrefix(line))
		if counts[normalized] == 0 {
			order = append(order, normalized)
		}
		counts[normalized]++
	}

	items := make([]grouped, 0, len(order))
	for _, msg := range order {
		items = append(items, grouped{Message: msg, Count: counts[msg]})
	}
	sort.Slice(items, func(i, j int) bool { return items[i].Count > items[j].Count })
	if len(items) > 25 {
		items = items[:25]
	}

	return map[string]any{
		"since":          since,
		"total_raw_rows": totalRaw,
		"unique_errors":  len(order),
		"entries":        items,
	}, nil
}

func stripJournalPrefix(line string) string {
	idx := strings.Index(line, ": ")
	if idx > 0 && idx < 80 {
		return line[idx+2:]
	}
	return line
}

func runNetPing(ctx context.Context, args map[string]string) (any, error) {

// EnsureConfigFile создаёт шаблон llm.jsonc с правами 0600, если он ещё не существует.
func EnsureConfigFile(path string) error {
	if _, err := os.Stat(path); err == nil {
		_ = os.Chmod(path, 0o600)
		return nil
	}
	if err := os.MkdirAll(filepath.Dir(path), 0o700); err != nil {
		return err
	}
	return os.WriteFile(path, []byte(defaultConfigTemplate), 0o600)
}

// NewRouter создаёт и загружает каскадный роутер моделей.
func NewRouter(path string) (*Router, error) {
	if strings.TrimSpace(path) == "" {
		path = DefaultConfigPath()
	}
	if err := EnsureConfigFile(path); err != nil {
		return nil, err
	}
	r := &Router{
		cfgPath:   path,
		cooldowns: make(map[string]time.Time),
	}
	if err := r.Reload(); err != nil {
		return nil, err
	}
	return r, nil
}

// ConfigPath возвращает путь к используемому файлу конфигурации ИИ.
func (r *Router) ConfigPath() string { return r.cfgPath }

// Reload перечитывает llm.jsonc с диска на лету без перезапуска демона.
func (r *Router) Reload() error {
	raw, err := os.ReadFile(r.cfgPath)
	if err != nil {
		return err
	}
	var cfg Config
	if err := json.Unmarshal(stripJSONC(raw), &cfg); err != nil {
		return fmt.Errorf("ошибка чтения %s: %w", r.cfgPath, err)
	}
	if cfg.ActiveProfile == "" {
		cfg.ActiveProfile = "auto"
	}
	if cfg.CooldownSec <= 0 {
		cfg.CooldownSec = defaultCooldownSec
	}
	r.mu.Lock()
	r.cfg = cfg
	r.mu.Unlock()
	return nil
}

// ResetHistory очищает контекст текущего диалога в консоли.
func (r *Router) ResetHistory() {
	r.mu.Lock()
	r.history = nil
	r.mu.Unlock()
}

// ProfileStatus возвращает сводку по всем профилям (без раскрытия ключей!).
type ProfileStatus struct {
	ID            string
	Title         string
	Model         string
	HasKey        bool
	CoolingSecRem int
}

// Status возвращает текущий режим выбора, путь к конфигу и список профилей.
func (r *Router) Status() (active string, readyCount int, items []ProfileStatus) {
	r.mu.Lock()
	defer r.mu.Unlock()

	now := time.Now()
	active = r.cfg.ActiveProfile
	for _, p := range r.cfg.Profiles {
		hasKey := r.resolveKeyLocked(p) != ""
		if hasKey {
			readyCount++
		}
		rem := 0
		if until, ok := r.cooldowns[p.ID]; ok && until.After(now) {
			rem = int(until.Sub(now).Seconds())
		}
		items = append(items, ProfileStatus{
			ID:            p.ID,
			Title:         p.Title,
			Model:         p.Model,
			HasKey:        hasKey,
			CoolingSecRem: rem,
		})
	}
	return active, readyCount, items
}

// SetActiveProfile переключает активную модель ("auto" или конкретный ID профиля) в памяти.

	defer r.mu.Unlock()

	if id == "auto" {
		r.cfg.ActiveProfile = "auto"
		return nil
	}
	for _, p := range r.cfg.Profiles {
		if p.ID == id {
			r.cfg.ActiveProfile = id
			return nil
		}
	}
	return fmt.Errorf("профиль %q не найден", id)
}

func (r *Router) resolveKeyLocked(p Profile) string {
	if strings.TrimSpace(p.APIKey) != "" {
		return strings.TrimSpace(p.APIKey)
	}
	if p.KeyRef != "" && r.cfg.Keys != nil {
		if v := strings.TrimSpace(r.cfg.Keys[p.KeyRef]); v != "" {
			return v
		}
	}
	if p.APIKeyEnv != "" {
		if v := strings.TrimSpace(os.Getenv(p.APIKeyEnv)); v != "" {
			return v
		}
	}
	if strings.Contains(p.BaseURL, "127.0.0.1") || strings.Contains(p.BaseURL, "localhost") || strings.Contains(p.BaseURL, "11434") {
		return "ollama-local"
	}
	return ""
}

func (r *Router) candidateOrder() ([]Profile, map[string]string) {
	r.mu.Lock()
	defer r.mu.Unlock()

	now := time.Now()
	keys := make(map[string]string)
	var fresh []Profile
	var cooling []Profile

	push := func(p Profile) {
		k := r.resolveKeyLocked(p)
		if k == "" {
			return
		}
		keys[p.ID] = k
		if until, ok := r.cooldowns[p.ID]; ok && until.After(now) {
			cooling = append(cooling, p)
		} else {
			fresh = append(fresh, p)
		}
	}

	if r.cfg.ActiveProfile != "" && r.cfg.ActiveProfile != "auto" {
		for _, p := range r.cfg.Profiles {
			if p.ID == r.cfg.ActiveProfile {
				push(p)
			}
		}
		if r.cfg.Fallback {
			for _, p := range r.cfg.Profiles {
				if p.ID != r.cfg.ActiveProfile {
					push(p)
				}
			}
		}
	} else {
		for _, p := range r.cfg.Profiles {
			push(p)
		}
	}

	return append(fresh, cooling...), keys
}

func (r *Router) markCooldown(profileID string) int {
	r.mu.Lock()
	defer r.mu.Unlock()
	sec := r.cfg.CooldownSec
	if sec <= 0 {
		sec = defaultCooldownSec
	}
	r.cooldowns[profileID] = time.Now().Add(time.Duration(sec) * time.Second)
	return sec
}

	StatusCode int
	Retryable  bool

// Ask отправляет вопрос пользователя в каскад моделей, выполняет разрешённые инструменты ядра и возвращает ответ.
func (r *Router) Ask(
	ctx context.Context,
	userQuestion string,
	specs []core.ToolSpec,
	execTool ToolExecutor,
	onEvent EventLogger,
) (answer string, usedProfile Profile, err error) {
	_ = r.Reload()

	candidates, keyMap := r.candidateOrder()
	if len(candidates) == 0 {
		return "", Profile{}, ErrNoKeys
	}

	apiTools, llmToReal := encodeTools(specs)

	r.mu.Lock()
	baseHistory := make([]Message, len(r.history))
	copy(baseHistory, r.history)
	r.mu.Unlock()

	userMsg := Message{Role: "user", Content: userQuestion}
	turnMessages := append(baseHistory, userMsg)

	idx := 0
	current := candidates[idx]

	for round := 0; round < maxToolRounds; round++ {
		var assistantMsg Message
		var callErr error

		for idx < len(candidates) {
			current = candidates[idx]
			assistantMsg, callErr = callCompletionAPI(ctx, current, keyMap[current.ID], turnMessages, apiTools)
			if callErr == nil {
				break
			}
			var ae *apiError
			if errors.As(callErr, &ae) && ae.Retryable && idx+1 < len(candidates) {
				sec := r.markCooldown(current.ID)
				idx++
				next := candidates[idx]
				if onEvent != nil {
					onEvent(fmt.Sprintf("⚡ Лимит/сбой у %s (HTTP %d, пауза %dс) → авто-переключение на %s",
						current.ID, ae.StatusCode, sec, next.Title))
				}
				continue
			}
			return "", current, callErr
		}

		if len(assistantMsg.ToolCalls) == 0 {
			finalText := strings.TrimSpace(assistantMsg.Content)
			turnMessages = append(turnMessages, Message{Role: "assistant", Content: finalText})
			r.saveHistory(turnMessages)
			return finalText, current, nil
		}

		turnMessages = append(turnMessages, assistantMsg)

		for _, tc := range assistantMsg.ToolCalls {
			realName, ok := llmToReal[tc.Function.Name]
			if !ok {
				realName = strings.ReplaceAll(tc.Function.Name, "__", ".")
			}
			parsedArgs := parseToolArguments(tc.Function.Arguments)

			if onEvent != nil {
				onEvent(fmt.Sprintf("↳ ИИ вызывает %s(%s)", realName, formatShortArgs(parsedArgs)))
			}

			toolOutput, execErr := execTool(ctx, realName, parsedArgs)
			if execErr != nil {
				toolOutput = fmt.Sprintf(`{"ok":false,"error":%q}`, execErr.Error())
			}

			turnMessages = append(turnMessages, Message{
				Role:       "tool",
				ToolCallID: tc.ID,
				Content:    toolOutput,
			})
		}
	}

	return "", current, fmt.Errorf("превышен лимит шагов вызова инструментов (%d)", maxToolRounds)
}

func (r *Router) saveHistory(msgs []Message) {
	var compact []Message
	for _, m := range msgs {
		if (m.Role == "user" || m.Role == "assistant") && len(m.ToolCalls) == 0 && strings.TrimSpace(m.Content) != "" {
			compact = append(compact, Message{Role: m.Role, Content: m.Content})
		}
	}
	if len(compact) > maxHistoryMessages {
		compact = compact[len(compact)-maxHistoryMessages:]
	}
	r.mu.Lock()
	r.history = compact
	r.mu.Unlock()
}

type chatRequest struct {
	Model       string           `json:"model"`
	Messages    []Message        `json:"messages"`
	Tools       []map[string]any `json:"tools,omitempty"`
	Temperature float64          `json:"temperature"`
}

type chatResponse struct {
	Choices []struct {
		Message Message `json:"message"`
	} `json:"choices"`
	Error *struct {
		Message string `json:"message"`
	} `json:"error,omitempty"`
}

func callCompletionAPI(
	ctx context.Context,
	p Profile,
	apiKey string,
	history []Message,
	tools []map[string]any,
) (Message, error) {
	timeoutSec := p.TimeoutSec
	if timeoutSec <= 0 {
		timeoutSec = defaultTimeoutSec
	}

	transport := &http.Transport{}
	if strings.TrimSpace(p.ProxyURL) != "" {
		u, err := url.Parse(strings.TrimSpace(p.ProxyURL))
		if err != nil {
			return Message{}, fmt.Errorf("некорректный proxy_url в профиле %s: %w", p.ID, err)
		}
		transport.Proxy = http.ProxyURL(u)
	}
	client := &http.Client{
		Timeout:   time.Duration(timeoutSec) * time.Second,
		Transport: transport,
	}

	wireMsgs := make([]Message, 0, len(history)+1)
	wireMsgs = append(wireMsgs, Message{Role: "system", Content: systemPromptRU})
	wireMsgs = append(wireMsgs, history...)

	payload := chatRequest{
		Model:       p.Model,
		Messages:    wireMsgs,
		Tools:       tools,
		Temperature: 0.2,
	}
	bodyBytes, err := json.Marshal(payload)
	if err != nil {
		return Message{}, err
	}

	reqCtx, cancel := context.WithTimeout(ctx, time.Duration(timeoutSec)*time.Second)
	defer cancel()

	endpoint := buildCompletionsURL(p.BaseURL)
	httpReq, err := http.NewRequestWithContext(reqCtx, http.MethodPost, endpoint, bytes.NewReader(bodyBytes))
	if err != nil {
		return Message{}, err
	}
	httpReq.Header.Set("Content-Type", "application/json")
	httpReq.Header.Set("Accept", "application/json")
	httpReq.Header.Set("Authorization", "Bearer "+apiKey)

	resp, err := client.Do(httpReq)
	if err != nil {
		return Message{}, &apiError{
			StatusCode: 0,
			Retryable:  true,
			Message:    fmt.Sprintf("сетевой сбой %s: %v", p.ID, err),
		}
	}
	defer resp.Body.Close()

	respBytes, _ := io.ReadAll(io.LimitReader(resp.Body, 2<<20))
	if resp.StatusCode >= 300 {
		retryable := resp.StatusCode == 429 || resp.StatusCode == 408 || resp.StatusCode >= 500 ||
			resp.StatusCode == 401 || resp.StatusCode == 403
		shortBody := strings.TrimSpace(string(respBytes))
		if len(shortBody) > 220 {
			shortBody = shortBody[:220] + "..."
		}
		return Message{}, &apiError{
			StatusCode: resp.StatusCode,
			Retryable:  retryable,
			Message:    fmt.Sprintf("ошибка API %s (HTTP %d): %s", p.ID, resp.StatusCode, shortBody),
		}
	}

	var parsed chatResponse
	if err := json.Unmarshal(respBytes, &parsed); err != nil {
		return Message{}, fmt.Errorf("некорректный JSON от %s: %w", p.ID, err)
	}
	if len(parsed.Choices) == 0 {
		return Message{}, &apiError{
			StatusCode: resp.StatusCode,
			Retryable:  true,
			Message:    fmt.Sprintf("пустой ответ choices от %s", p.ID),
		}
	}
	msg := parsed.Choices[0].Message
	if msg.Role == "" {
		msg.Role = "assistant"
	}
	return msg, nil
}

func buildCompletionsURL(base string) string {
	b := strings.TrimRight(strings.TrimSpace(base), "/")
	if strings.HasSuffix(b, "/chat/completions") {
		return b
	}
	return b + "/chat/completions"
}

func encodeTools(specs []core.ToolSpec) ([]map[string]any, map[string]string) {
	var out []map[string]any
	mapping := make(map[string]string, len(specs))

	for _, sp := range specs {
		if sp.Name == "core.tools" {
			continue
		}
		safeName := strings.ReplaceAll(sp.Name, ".", "__")
		mapping[safeName] = sp.Name

		out = append(out, map[string]any{
			"type": "function",
			"function": map[string]any{
				"name":        safeName,
				"description": fmt.Sprintf("[%s] %s", sp.Risk, sp.Description),
				"parameters":  sp.InputJSONSchema(),
			},
		})
	}
	return out, mapping
}

func parseToolArguments(raw string) map[string]string {
	out := make(map[string]string)
	raw = strings.TrimSpace(raw)
	if raw == "" || raw == "{}" {
		return out
	}
	var generic map[string]any
	if err := json.Unmarshal([]byte(raw), &generic); err != nil {
		return out
	}
	for k, v := range generic {
		if v == nil {
			continue
		}
		out[k] = fmt.Sprint(v)
	}
	return out
}

func formatShortArgs(args map[string]string) string {
	if len(args) == 0 {
		return ""
	}
	parts := make([]string, 0, len(args))
	for k, v := range args {
		parts = append(parts, fmt.Sprintf("%s=%s", k, v))
	}
	return strings.Join(parts, ", ")
}

func stripJSONC(src []byte) []byte {
	out := make([]byte, 0, len(src))
	inStr := false
	escaped := false

	for i := 0; i < len(src); i++ {
		c := src[i]
		if inStr {
			out = append(out, c)
			if escaped {
				escaped = false
				continue
			}
			if c == '\\' {
				escaped = true
			} else if c == '"' {
				inStr = false
			}
			continue
		}
		if c == '"' {
			inStr = true
			out = append(out, c)
			continue
		}
		if c == '/' && i+1 < len(src) && src[i+1] == '/' {
			i += 2
			for i < len(src) && src[i] != '\n' {
				i++
			}
			if i < len(src) {
				out = append(out, '\n')
			}
			continue
		}
		if c == '/' && i+1 < len(src) && src[i+1] == '*' {
			i += 2
			for i+1 < len(src) && !(src[i] == '*' && src[i+1] == '/') {
				i++
			}
			i++
			continue
		}
		out = append(out, c)
	}
	return out
}

			printAuditTail(cfg.Audit.Path, 10)

		case "log", "daemon-log":
			printLogTail(ipc.DaemonLogPath(cfg), 15)

		case "model", "models", "/model":
			handleModelMenu(router, routerErr, reader)

		case "mode", "/mode":
			handleToggleMode(&cfg, cfgPath)

		case "reset", "clear", "/reset":
			if router != nil {
				router.ResetHistory()
				fmt.Println("✓ История диалога с ИИ очищена.")
			}

		default:
			dispatchSmartInput(&cfg, cfgPath, router, routerErr, reader, line, cmd, rest)
		}

		if err == io.EOF {
			fmt.Println()
			return 0
		}
	}
}

func dispatchSmartInput(
	cfg *config.Config,
	cfgPath string,
	router *llm.Router,
	routerErr error,
	reader *bufio.Reader,
	fullLine, cmd string,
	rest []string,
) {
	if looksLikeRawDanger(fullLine) {
		execConsoleTool(*cfg, reader, fullLine, cmd, rest)
		return
	}

	specs, daemonUp := fetchDaemonSpecs(*cfg)
	if daemonUp && matchesRegisteredTool(specs, cmd, rest) {
		execConsoleTool(*cfg, reader, fullLine, cmd, rest)
		return
	}

	handleLLMChat(cfg, cfgPath, router, routerErr, reader, fullLine)
}

func handleLLMChat(
	cfg *config.Config,
	cfgPath string,
	router *llm.Router,
	routerErr error,
	reader *bufio.Reader,
	question string,
) {
	if routerErr != nil || router == nil {
		fmt.Fprintf(os.Stderr, "✗ Модуль ИИ не инициализирован: %v\n", routerErr)
		return
	}

	up, _ := ipc.IsRunningFast(*cfg)
	if !up {
		fmt.Println("• Демон остановлен — запускаю его для доступа к инструментам диагностики...")
		if _, _, err := ipc.StartDaemon(*cfg, cfgPath); err != nil {
			fmt.Fprintf(os.Stderr, "✗ Не удалось запустить демона: %v\n", err)
			return
		}
	}

	specs, ok := fetchDaemonSpecs(*cfg)
	if !ok {
		fmt.Fprintln(os.Stderr, "✗ Не удалось получить список инструментов от демона.")
		return
	}

	timeout := time.Duration(cfg.Policy.CommandTimeoutSec+5) * time.Second

	execViaDaemon := func(_ context.Context, toolName string, args map[string]string) (string, error) {
		req := core.Request{
			Tool:   toolName,
			Args:   args,
			Caller: "llm",
		}
		resp, err := ipc.CallUnix(cfg.SocketPath, req, timeout)
		if err != nil {
			return "", err
		}

		if !resp.OK && resp.NeedsApproval {
			if promptConfirmationReader(reader, resp.Tool, string(resp.Risk), formatArgsInline(args), resp.Error) {
				req.Approved = true
				resp, err = ipc.CallUnix(cfg.SocketPath, req, timeout)
				if err != nil {
					return "", err
				}
			} else {
				fmt.Println("  • Действие отклонено администратором.")
				return `{"ok":false,"approved":false,"note":"Отклонено администратором. Не повторяй вызов, дай безопасную команду для ручного выполнения."}`, nil
			}
		}

		if !resp.OK {
			return fmt.Sprintf(`{"ok":false,"policy_mode":%q,"error":%q}`, cfg.Policy.Mode, resp.Error), nil
		}
		return string(resp.Data), nil
	}

	fmt.Println("⏳ Запрос к ИИ...")
	ctx, cancel := context.WithTimeout(context.Background(), 2*time.Minute)
	defer cancel()

	answer, prof, err := router.Ask(ctx, question, specs, execViaDaemon, func(ev string) {
		fmt.Println(" ", ev)
	})
	if errors.Is(err, llm.ErrNoKeys) {
		printNoKeysHelp(router.ConfigPath())
		return
	}
	if err != nil {
		fmt.Fprintf(os.Stderr, "✗ Ошибка ИИ [%s]: %v\n", prof.ID, err)
		return
	}

	fmt.Printf("\n🤖 [%s]\n%s\n", prof.Title, answer)
}

func handleModelMenu(router *llm.Router, routerErr error, reader *bufio.Reader) {
	if routerErr != nil || router == nil {
		fmt.Fprintf(os.Stderr, "✗ Ошибка конфигурации ИИ: %v\n", routerErr)
		return
	}
	_ = router.Reload()
	active, readyCount, items := router.Status()

	fmt.Printf("\n--- Профили ИИ (%s) ---\n", router.ConfigPath())
	autoMark := " "
	if active == "auto" {
		autoMark = "*"
	}
	fmt.Printf("  [0] [%s] auto — Автоматический каскад (объединяет лимиты всех готовых моделей)\n", autoMark)
	for i, it := range items {

		}

		}
		fmt.Printf("  [%d] [%s] %-17s — %s [%s%s]\n", i+1, mark, it.ID, it.Title, keyBadge, coolInfo)

func fetchDaemonSpecs(cfg config.Config) ([]core.ToolSpec, bool) {
	resp, err := ipc.CallUnix(cfg.SocketPath, core.Request{
		Tool:   "core.tools",
		Caller: "status",
	}, 3*time.Second)
	if err != nil || !resp.OK {
		return nil, false
	}
	var payload struct {
		Tools []core.ToolSpec `json:"tools"`
	}
	if err := json.Unmarshal(resp.Data, &payload); err != nil {
		return nil, false
	}
	return payload.Tools, true
}

func matchesRegisteredTool(specs []core.ToolSpec, cmd string, rest []string) bool {
	found := false
	for _, sp := range specs {
		short := sp.Name
		if idx := strings.IndexByte(short, '.'); idx >= 0 {
			short = short[idx+1:]
		}
		if cmd == sp.Name || cmd == short {
			found = true
			break
		}
	}
	if !found {
		return false
	}
	for _, arg := range rest {
		if strings.IndexByte(arg, '=') <= 0 {
			return false
		}
	}
	return true
}

func looksLikeRawDanger(line string) bool {
	low := strings.ToLower(strings.TrimSpace(line))
	prefixes := []string{
		"rm ", "mkfs", "dd if=", "chmod -r 777", ":(){", "shutdown", "reboot", "poweroff",
	}
	for _, p := range prefixes {
		if strings.HasPrefix(low, p) {
			return true
		}
	}
	return false
}

func handleServiceControl(action string, cfg config.Config, cfgPath string) int {
	switch action {
	case "start":
		st, already, err := ipc.StartDaemon(cfg, cfgPath)
		if err != nil {
			fmt.Fprintf(os.Stderr, "✗ Ошибка запуска: %v\n", err)
			return 1
		}
		if already {
			fmt.Printf("• Демон уже запущен (pid=%d, uptime=%s, память=%.1f МБ)\n",
				st.PID, st.Uptime, float64(st.RSSKB)/1024.0)
			return 0
		}
		fmt.Printf("✓ Демон запущен (pid=%d, режим=%s, память=%.1f МБ)\n",
			st.PID, st.PolicyMode, float64(st.RSSKB)/1024.0)
		return 0

	case "stop":
		pid, wasRunning, err := ipc.StopDaemon(cfg)
		if err != nil {
			fmt.Fprintf(os.Stderr, "✗ Ошибка остановки: %v\n", err)
			return 1
		}
		if !wasRunning {
			fmt.Println("• Демон уже остановлен.")
			return 0
		}
		fmt.Printf("✓ Демон штатно остановлен (pid=%d)\n", pid)
		return 0

	case "restart":
		oldPID, st, err := ipc.RestartDaemon(cfg, cfgPath)
		if err != nil {
			fmt.Fprintf(os.Stderr, "✗ Ошибка перезапуска: %v\n", err)
			return 1
		}
		if oldPID > 0 {
			fmt.Printf("✓ Старый процесс остановлен (pid=%d)\n", oldPID)
		}
		fmt.Printf("✓ Демон перезапущен (новый pid=%d, режим=%s, память=%.1f МБ)\n",
			st.PID, st.PolicyMode, float64(st.RSSKB)/1024.0)
		return 0

	case "status":
		st := ipc.GetStatus(cfg)
		printStatusSummary(st)
		if !st.Running {
			return 3
		}
		return 0
	}
	return 2
}

func execConsoleTool(cfg config.Config, reader *bufio.Reader, fullLine, toolName string, rawArgs []string) {
	kv := make(map[string]string)
	for _, item := range rawArgs {
		idx := strings.IndexByte(item, '=')
		if idx <= 0 {
			toolName = fullLine
			kv = nil
			break
		}
		kv[strings.TrimSpace(item[:idx])] = strings.TrimSpace(item[idx+1:])
	}

	req := core.Request{
		Tool:   toolName,
		Args:   kv,
		Caller: "console",
	}

	timeout := time.Duration(cfg.Policy.CommandTimeoutSec+5) * time.Second
	resp, err := ipc.CallUnix(cfg.SocketPath, req, timeout)
	if err != nil {
		fmt.Fprintf(os.Stderr, "✗ Демон не запущен (%v). Введите '1' или 'start', чтобы запустить его.\n", err)
		return
	}

	if !resp.OK && resp.NeedsApproval {
		if promptConfirmationReader(reader, resp.Tool, string(resp.Risk), formatArgsInline(kv), resp.Error) {
			req.Approved = true
			resp, err = ipc.CallUnix(cfg.SocketPath, req, timeout)
			if err != nil {
				fmt.Fprintf(os.Stderr, "✗ Ошибка повторного запроса: %v\n", err)
				return
			}
		} else {
			fmt.Println("• Операция отменена.")
			return
		}
	}

	if !resp.OK {
		fmt.Fprintf(os.Stderr, "✗ Отказ политики [%s]: %s\n", resp.Tool, resp.Error)
		return
	}

	var pretty any
	if err := json.Unmarshal(resp.Data, &pretty); err != nil {
		fmt.Println(string(resp.Data))
		return
	}
	out, _ := json.MarshalIndent(pretty, "", "  ")
	fmt.Println(string(out))
}

func promptConfirmationReader(reader *bufio.Reader, toolName, risk, argsStr, reason string) bool {
	fmt.Printf("\n⚠️  Требуется подтверждение\n   Операция:  %s (риск: %s)\n", toolName, risk)
	if argsStr != "" {
		fmt.Printf("   Параметры: %s\n", argsStr)
	}
	fmt.Printf("   Причина:   %s\n   Выполнить? [y/N]: ", reason)

	line, err := reader.ReadString('\n')
	if err != nil && len(line) == 0 {
		return false
	}
	ans := strings.ToLower(strings.TrimSpace(line))
	return ans == "y" || ans == "yes" || ans == "д" || ans == "да"
}

func formatArgsInline(args map[string]string) string {
	if len(args) == 0 {
		return ""
	}
	parts := make([]string, 0, len(args))
	for k, v := range args {
		parts = append(parts, k+"="+v)
	}
	return strings.Join(parts, ", ")
}

func splitConsoleLine(input string) ([]string, error) {
	var tokens []string
	var buf strings.Builder
	var quote rune
	inToken := false

	for _, r := range input {
		switch {
		case quote != 0:
			if r == quote {
				quote = 0
			} else {
				buf.WriteRune(r)
			}
		case r == '"' || r == '\'':
			quote = r
			inToken = true
		case unicode.IsSpace(r):
			if inToken {
				tokens = append(tokens, buf.String())
				buf.Reset()
				inToken = false
			}
		default:
			buf.WriteRune(r)
			inToken = true
		}
	}
	if quote != 0 {
		return nil, io.ErrUnexpectedEOF
	}
	if inToken {
		tokens = append(tokens, buf.String())
	}
	return tokens, nil
}

func dispatchSmartInput(
	cfg *config.Config,
	cfgPath string,
	router *llm.Router,
	routerErr error,
	reader *bufio.Reader,
	fullLine, cmd string,
	rest []string,
) {
	if looksLikeRawDanger(fullLine) {
		execConsoleTool(*cfg, reader, fullLine, cmd, rest)
		return
	}

	specs, daemonUp := fetchDaemonSpecs(*cfg)
	if daemonUp && matchesRegisteredTool(specs, cmd, rest) {
		execConsoleTool(*cfg, reader, fullLine, cmd, rest)
		return
	}

	handleLLMChat(cfg, cfgPath, router, routerErr, reader, fullLine)
}

func handleLLMChat(
	cfg *config.Config,
	cfgPath string,
	router *llm.Router,
	routerErr error,
	reader *bufio.Reader,
	question string,
) {
	if routerErr != nil || router == nil {
		fmt.Fprintf(os.Stderr, "✗ Модуль ИИ не инициализирован: %v\n", routerErr)
		return
	}

	up, _ := ipc.IsRunningFast(*cfg)
	if !up {
		fmt.Println("• Демон остановлен — запускаю его для доступа к инструментам диагностики...")
		if _, _, err := ipc.StartDaemon(*cfg, cfgPath); err != nil {
			fmt.Fprintf(os.Stderr, "✗ Не удалось запустить демона: %v\n", err)
			return
		}
	}

	specs, ok := fetchDaemonSpecs(*cfg)
	if !ok {
		fmt.Fprintln(os.Stderr, "✗ Не удалось получить список инструментов от демона.")
		return
	}

	timeout := time.Duration(cfg.Policy.CommandTimeoutSec+5) * time.Second

	execViaDaemon := func(_ context.Context, toolName string, args map[string]string) (string, error) {
		req := core.Request{
			Tool:   toolName,
			Args:   args,
			Caller: "llm",
		}
		resp, err := ipc.CallUnix(cfg.SocketPath, req, timeout)
		if err != nil {
			return "", err
		}

		if !resp.OK && resp.NeedsApproval {
			if promptConfirmationReader(reader, resp.Tool, string(resp.Risk), formatArgsInline(args), resp.Error) {
				req.Approved = true
				resp, err = ipc.CallUnix(cfg.SocketPath, req, timeout)
				if err != nil {
					return "", err
				}
			} else {
				fmt.Println("  • Действие отклонено администратором.")
				return `{"ok":false,"approved":false,"note":"Отклонено администратором. Не повторяй вызов, дай безопасную команду для ручного выполнения."}`, nil
			}
		}

		if !resp.OK {
			return fmt.Sprintf(`{"ok":false,"policy_mode":%q,"error":%q}`, cfg.Policy.Mode, resp.Error), nil
		}
		return string(resp.Data), nil
	}

	fmt.Println("⏳ Запрос к ИИ...")
	ctx, cancel := context.WithTimeout(context.Background(), 2*time.Minute)
	defer cancel()

	answer, prof, err := router.Ask(ctx, question, specs, execViaDaemon, func(ev string) {
		fmt.Println(" ", ev)
	})
	if errors.Is(err, llm.ErrNoKeys) {
		printNoKeysHelp(router.ConfigPath())
		return
	}
	if err != nil {
		fmt.Fprintf(os.Stderr, "✗ Ошибка ИИ [%s]: %v\n", prof.ID, err)
		return
	}

	fmt.Printf("\n🤖 [%s]\n%s\n", prof.Title, answer)
}

func handleModelMenu(router *llm.Router, routerErr error, reader *bufio.Reader) {
	if routerErr != nil || router == nil {
		fmt.Fprintf(os.Stderr, "✗ Ошибка конфигурации ИИ: %v\n", routerErr)
		return
	}
	_ = router.Reload()
	active, readyCount, items := router.Status()

	fmt.Printf("\n--- Профили ИИ (%s) ---\n", router.ConfigPath())
	autoMark := " "
	if active == "auto" {
		autoMark = "*"
	}
	fmt.Printf("  [0] [%s] auto — Автоматический каскад (объединяет лимиты всех готовых моделей)\n", autoMark)
	for i, it := range items {
		mark := " "
		if active == it.ID {
			mark = "*"
		}
		keyBadge := "нет ключа"
		if it.HasKey {
			keyBadge = "ключ OK"
		}
		coolInfo := ""
		if it.CoolingSecRem > 0 {
			coolInfo = fmt.Sprintf(", пауза %dс", it.CoolingSecRem)
		}
		fmt.Printf("  [%d] [%s] %-17s — %s [%s%s]\n", i+1, mark, it.ID, it.Title, keyBadge, coolInfo)
	}
	if readyCount == 0 {
		printNoKeysHelp(router.ConfigPath())
		return
	}
	fmt.Print("\nВведите номер профиля (0 = auto, Enter = оставить как есть): ")
	choiceLine, _ := reader.ReadString('\n')
	choiceLine = strings.TrimSpace(choiceLine)
	if choiceLine == "" {
		return
	}
	if choiceLine == "0" || choiceLine == "auto" {
		_ = router.SetActiveProfile("auto")
		fmt.Println("✓ Включён режим 'auto' (автоматический каскад моделей).")
		return
	}
	if n, err := strconv.Atoi(choiceLine); err == nil && n >= 1 && n <= len(items) {
		picked := items[n-1]
		_ = router.SetActiveProfile(picked.ID)
		fmt.Printf("✓ Выбрана основная модель: %s (%s)\n", picked.Title, picked.ID)
		return
	}
	if err := router.SetActiveProfile(choiceLine); err == nil {
		fmt.Printf("✓ Выбрана основная модель: %s\n", choiceLine)
		return
	}
	fmt.Println("• Профиль не изменён.")
}

func handleToggleMode(cfg *config.Config, cfgPath string) {
	next := "ask"
	desc := "чтение — автоматически, изменения — строго с вашим подтверждением [y/N]"
	if cfg.Policy.Mode == "ask" {
		next = "readonly"
		desc = "строго только чтение (любые изменения заблокированы политикой)"
	}
	cfg.Policy.Mode = next

	up, _ := ipc.IsRunningFast(*cfg)
	if up {
		_, st, err := ipc.RestartDaemon(*cfg, cfgPath)
		if err != nil {
			fmt.Fprintf(os.Stderr, "✗ Ошибка перезапуска демона в новом режиме: %v\n", err)
			return
		}
		fmt.Printf("✓ Режим переключён на '%s' (%s; новый pid=%d)\n", st.PolicyMode, desc, st.PID)
		return
	}
	fmt.Printf("✓ Режим для следующего запуска переключён на '%s' (%s)\n", next, desc)
}

func fetchDaemonSpecs(cfg config.Config) ([]core.ToolSpec, bool) {
	resp, err := ipc.CallUnix(cfg.SocketPath, core.Request{
		Tool:   "core.tools",
		Caller: "status",
	}, 3*time.Second)
	if err != nil || !resp.OK {
		return nil, false
	}
	var payload struct {
		Tools []core.ToolSpec `json:"tools"`
	}
	if err := json.Unmarshal(resp.Data, &payload); err != nil {
		return nil, false
	}
	return payload.Tools, true
}

func matchesRegisteredTool(specs []core.ToolSpec, cmd string, rest []string) bool {
	found := false
	for _, sp := range specs {
		short := sp.Name
		if idx := strings.IndexByte(short, '.'); idx >= 0 {
			short = short[idx+1:]
		}
		if cmd == sp.Name || cmd == short {
			found = true
			break
		}
	}
	if !found {
		return false
	}
	for _, arg := range rest {
		if strings.IndexByte(arg, '=') <= 0 {
			return false
		}
	}
	return true
}

func looksLikeRawDanger(line string) bool {
	low := strings.ToLower(strings.TrimSpace(line))
	prefixes := []string{
		"rm ", "mkfs", "dd if=", "chmod -r 777", ":(){", "shutdown", "reboot", "poweroff",
	}
	for _, p := range prefixes {
		if strings.HasPrefix(low, p) {
			return true
		}
	}
	return false
}

func handleServiceControl(action string, cfg config.Config, cfgPath string) int {
	switch action {
	case "start":
		st, already, err := ipc.StartDaemon(cfg, cfgPath)
		if err != nil {
			fmt.Fprintf(os.Stderr, "✗ Ошибка запуска: %v\n", err)
			return 1
		}
		if already {
			fmt.Printf("• Демон уже запущен (pid=%d, uptime=%s, память=%.1f МБ)\n",
				st.PID, st.Uptime, float64(st.RSSKB)/1024.0)
			return 0
		}
		fmt.Printf("✓ Демон запущен (pid=%d, режим=%s, память=%.1f МБ)\n",
			st.PID, st.PolicyMode, float64(st.RSSKB)/1024.0)
		return 0

	case "stop":
		pid, wasRunning, err := ipc.StopDaemon(cfg)
		if err != nil {
			fmt.Fprintf(os.Stderr, "✗ Ошибка остановки: %v\n", err)
			return 1
		}
		if !wasRunning {
			fmt.Println("• Демон уже остановлен.")
			return 0
		}
		fmt.Printf("✓ Демон штатно остановлен (pid=%d)\n", pid)
		return 0

	case "restart":
		oldPID, st, err := ipc.RestartDaemon(cfg, cfgPath)
		if err != nil {
			fmt.Fprintf(os.Stderr, "✗ Ошибка перезапуска: %v\n", err)
			return 1
		}
		if oldPID > 0 {
			fmt.Printf("✓ Старый процесс остановлен (pid=%d)\n", oldPID)
		}
		fmt.Printf("✓ Демон перезапущен (новый pid=%d, режим=%s, память=%.1f МБ)\n",
			st.PID, st.PolicyMode, float64(st.RSSKB)/1024.0)
		return 0

	case "status":
		st := ipc.GetStatus(cfg)
		printStatusSummary(st)
		if !st.Running {
			return 3
		}
		return 0
	}
	return 2
}

func execConsoleTool(cfg config.Config, reader *bufio.Reader, fullLine, toolName string, rawArgs []string) {
	kv := make(map[string]string)
	for _, item := range rawArgs {
		idx := strings.IndexByte(item, '=')
		if idx <= 0 {
			toolName = fullLine
			kv = nil
			break
		}
		kv[strings.TrimSpace(item[:idx])] = strings.TrimSpace(item[idx+1:])
	}

	req := core.Request{
		Tool:   toolName,
		Args:   kv,
		Caller: "console",
	}

	timeout := time.Duration(cfg.Policy.CommandTimeoutSec+5) * time.Second
	resp, err := ipc.CallUnix(cfg.SocketPath, req, timeout)
	if err != nil {
		fmt.Fprintf(os.Stderr, "✗ Демон не запущен (%v). Введите '1' или 'start', чтобы запустить его.\n", err)
		return
	}

	if !resp.OK && resp.NeedsApproval {
		if promptConfirmationReader(reader, resp.Tool, string(resp.Risk), formatArgsInline(kv), resp.Error) {
			req.Approved = true
			resp, err = ipc.CallUnix(cfg.SocketPath, req, timeout)
			if err != nil {
				fmt.Fprintf(os.Stderr, "✗ Ошибка повторного запроса: %v\n", err)
				return
			}
		} else {
			fmt.Println("• Операция отменена.")
			return
		}
	}

	if !resp.OK {
		fmt.Fprintf(os.Stderr, "✗ Отказ политики [%s]: %s\n", resp.Tool, resp.Error)
		return
	}

	var pretty any
	if err := json.Unmarshal(resp.Data, &pretty); err != nil {
		fmt.Println(string(resp.Data))
		return
	}
	out, _ := json.MarshalIndent(pretty, "", "  ")
	fmt.Println(string(out))
}

func promptConfirmationReader(reader *bufio.Reader, toolName, risk, argsStr, reason string) bool {
	fmt.Printf("\n⚠️  Требуется подтверждение\n   Операция:  %s (риск: %s)\n", toolName, risk)
	if argsStr != "" {
		fmt.Printf("   Параметры: %s\n", argsStr)
	}
	fmt.Printf("   Причина:   %s\n   Выполнить? [y/N]: ", reason)

	line, err := reader.ReadString('\n')
	if err != nil && len(line) == 0 {
		return false
	}
	ans := strings.ToLower(strings.TrimSpace(line))
	return ans == "y" || ans == "yes" || ans == "д" || ans == "да"
}

func formatArgsInline(args map[string]string) string {
	if len(args) == 0 {
		return ""
	}
	parts := make([]string, 0, len(args))
	for k, v := range args {
		parts = append(parts, k+"="+v)
	}
	return strings.Join(parts, ", ")
}

func splitConsoleLine(input string) ([]string, error) {
	var tokens []string
	var buf strings.Builder
	var quote rune
	inToken := false

	for _, r := range input {
		switch {
		case quote != 0:
			if r == quote {
				quote = 0
			} else {
				buf.WriteRune(r)
			}
		case r == '"' || r == '\'':
			quote = r
			inToken = true
		case unicode.IsSpace(r):
			if inToken {
				tokens = append(tokens, buf.String())
				buf.Reset()
				inToken = false
			}
		default:
			buf.WriteRune(r)
			inToken = true
		}
	}
	if quote != 0 {
		return nil, io.ErrUnexpectedEOF
	}
	if inToken {
		tokens = append(tokens, buf.String())
	}
	return tokens, nil
}

func dispatchSmartInput(
	cfg *config.Config,
	cfgPath string,
	router *llm.Router,
	routerErr error,
	reader *bufio.Reader,
	fullLine, cmd string,
	rest []string,
) {
	if looksLikeRawDanger(fullLine) {
		execConsoleTool(*cfg, reader, fullLine, cmd, rest)
		return
	}

	specs, daemonUp := fetchDaemonSpecs(*cfg)
	if daemonUp && matchesRegisteredTool(specs, cmd, rest) {
		execConsoleTool(*cfg, reader, fullLine, cmd, rest)
		return
	}

	handleLLMChat(cfg, cfgPath, router, routerErr, reader, fullLine)
}

func handleLLMChat(
	cfg *config.Config,
	cfgPath string,
	router *llm.Router,
	routerErr error,
	reader *bufio.Reader,
	question string,
) {
	if routerErr != nil || router == nil {
		fmt.Fprintf(os.Stderr, "✗ Модуль ИИ не инициализирован: %v\n", routerErr)
		return
	}

	up, _ := ipc.IsRunningFast(*cfg)
	if !up {
		fmt.Println("• Демон остановлен — запускаю его для доступа к инструментам диагностики...")
		if _, _, err := ipc.StartDaemon(*cfg, cfgPath); err != nil {
			fmt.Fprintf(os.Stderr, "✗ Не удалось запустить демона: %v\n", err)
			return
		}
	}

	specs, ok := fetchDaemonSpecs(*cfg)
	if !ok {
		fmt.Fprintln(os.Stderr, "✗ Не удалось получить список инструментов от демона.")
		return
	}

	timeout := time.Duration(cfg.Policy.CommandTimeoutSec+5) * time.Second

	execViaDaemon := func(_ context.Context, toolName string, args map[string]string) (string, error) {
		req := core.Request{
			Tool:   toolName,
			Args:   args,
			Caller: "llm",
		}
		resp, err := ipc.CallUnix(cfg.SocketPath, req, timeout)
		if err != nil {
			return "", err
		}

		if !resp.OK && resp.NeedsApproval {
			if promptConfirmationReader(reader, resp.Tool, string(resp.Risk), formatArgsInline(args), resp.Error) {
				req.Approved = true
				resp, err = ipc.CallUnix(cfg.SocketPath, req, timeout)
				if err != nil {
					return "", err
				}
			} else {
				fmt.Println("  • Действие отклонено администратором.")
				return `{"ok":false,"approved":false,"note":"Отклонено администратором. Не повторяй вызов, дай безопасную команду для ручного выполнения."}`, nil
			}
		}

		if !resp.OK {
			return fmt.Sprintf(`{"ok":false,"policy_mode":%q,"error":%q}`, cfg.Policy.Mode, resp.Error), nil
		}
		return string(resp.Data), nil
	}

	fmt.Println("⏳ Запрос к ИИ...")
	ctx, cancel := context.WithTimeout(context.Background(), 2*time.Minute)
	defer cancel()

	answer, prof, err := router.Ask(ctx, question, specs, execViaDaemon, func(ev string) {
		fmt.Println(" ", ev)
	})
	if errors.Is(err, llm.ErrNoKeys) {
		printNoKeysHelp(router.ConfigPath())
		return
	}
	if err != nil {
		fmt.Fprintf(os.Stderr, "✗ Ошибка ИИ [%s]: %v\n", prof.ID, err)
		return
	}

	fmt.Printf("\n🤖 [%s]\n%s\n", prof.Title, answer)
}

func handleModelMenu(router *llm.Router, routerErr error, reader *bufio.Reader) {
	if routerErr != nil || router == nil {
		fmt.Fprintf(os.Stderr, "✗ Ошибка конфигурации ИИ: %v\n", routerErr)
		return
	}
	_ = router.Reload()
	active, readyCount, items := router.Status()

	fmt.Printf("\n--- Профили ИИ (%s) ---\n", router.ConfigPath())
	autoMark := " "
	if active == "auto" {
		autoMark = "*"
	}
	fmt.Printf("  [0] [%s] auto — Автоматический каскад (объединяет лимиты всех готовых моделей)\n", autoMark)
	for i, it := range items {
		mark := " "
		if active == it.ID {
			mark = "*"
		}
		keyBadge := "нет ключа"
		if it.HasKey {
			keyBadge = "ключ OK"
		}
		coolInfo := ""
		if it.CoolingSecRem > 0 {
			coolInfo = fmt.Sprintf(", пауза %dс", it.CoolingSecRem)
		}
		fmt.Printf("  [%d] [%s] %-17s — %s [%s%s]\n", i+1, mark, it.ID, it.Title, keyBadge, coolInfo)
	}
	if readyCount == 0 {
		printNoKeysHelp(router.ConfigPath())
		return
	}
	fmt.Print("\nВведите номер профиля (0 = auto, Enter = оставить как есть): ")
	choiceLine, _ := reader.ReadString('\n')
	choiceLine = strings.TrimSpace(choiceLine)
	if choiceLine == "" {
		return
	}
	if choiceLine == "0" || choiceLine == "auto" {
		_ = router.SetActiveProfile("auto")
		fmt.Println("✓ Включён режим 'auto' (автоматический каскад моделей).")
		return
	}
	if n, err := strconv.Atoi(choiceLine); err == nil && n >= 1 && n <= len(items) {
		picked := items[n-1]
		_ = router.SetActiveProfile(picked.ID)
		fmt.Printf("✓ Выбрана основная модель: %s (%s)\n", picked.Title, picked.ID)
		return
	}
	if err := router.SetActiveProfile(choiceLine); err == nil {
		fmt.Printf("✓ Выбрана основная модель: %s\n", choiceLine)
		return
	}
	fmt.Println("• Профиль не изменён.")
}

func handleToggleMode(cfg *config.Config, cfgPath string) {
	next := "ask"
	desc := "чтение — автоматически, изменения — строго с вашим подтверждением [y/N]"
	if cfg.Policy.Mode == "ask" {
		next = "readonly"
		desc = "строго только чтение (любые изменения заблокированы политикой)"
	}
	cfg.Policy.Mode = next

	up, _ := ipc.IsRunningFast(*cfg)
	if up {
		_, st, err := ipc.RestartDaemon(*cfg, cfgPath)
		if err != nil {
			fmt.Fprintf(os.Stderr, "✗ Ошибка перезапуска демона в новом режиме: %v\n", err)
			return
		}
		fmt.Printf("✓ Режим переключён на '%s' (%s; новый pid=%d)\n", st.PolicyMode, desc, st.PID)
		return
	}
	fmt.Printf("✓ Режим для следующего запуска переключён на '%s' (%s)\n", next, desc)
}

func fetchDaemonSpecs(cfg config.Config) ([]core.ToolSpec, bool) {
	resp, err := ipc.CallUnix(cfg.SocketPath, core.Request{
		Tool:   "core.tools",
		Caller: "status",
	}, 3*time.Second)
	if err != nil || !resp.OK {
		return nil, false
	}
	var payload struct {
		Tools []core.ToolSpec `json:"tools"`
	}
	if err := json.Unmarshal(resp.Data, &payload); err != nil {
		return nil, false
	}
	return payload.Tools, true
}

func matchesRegisteredTool(specs []core.ToolSpec, cmd string, rest []string) bool {
	found := false
	for _, sp := range specs {
		short := sp.Name
		if idx := strings.IndexByte(short, '.'); idx >= 0 {
			short = short[idx+1:]
		}
		if cmd == sp.Name || cmd == short {
			found = true
			break
		}
	}
	if !found {
		return false
	}
	for _, arg := range rest {
		if strings.IndexByte(arg, '=') <= 0 {
			return false
		}
	}
	return true
}

func looksLikeRawDanger(line string) bool {
	low := strings.ToLower(strings.TrimSpace(line))
	prefixes := []string{
		"rm ", "mkfs", "dd if=", "chmod -r 777", ":(){", "shutdown", "reboot", "poweroff",
	}
	for _, p := range prefixes {
		if strings.HasPrefix(low, p) {
			return true
		}
	}
	return false
}

func handleServiceControl(action string, cfg config.Config, cfgPath string) int {
	switch action {
	case "start":
		st, already, err := ipc.StartDaemon(cfg, cfgPath)
		if err != nil {
			fmt.Fprintf(os.Stderr, "✗ Ошибка запуска: %v\n", err)
			return 1
		}
		if already {
			fmt.Printf("• Демон уже запущен (pid=%d, uptime=%s, память=%.1f МБ)\n",
				st.PID, st.Uptime, float64(st.RSSKB)/1024.0)
			return 0
		}
		fmt.Printf("✓ Демон запущен (pid=%d, режим=%s, память=%.1f МБ)\n",
			st.PID, st.PolicyMode, float64(st.RSSKB)/1024.0)
		return 0

	case "stop":
		pid, wasRunning, err := ipc.StopDaemon(cfg)
		if err != nil {
			fmt.Fprintf(os.Stderr, "✗ Ошибка остановки: %v\n", err)
			return 1
		}
		if !wasRunning {
			fmt.Println("• Демон уже остановлен.")
			return 0
		}
		fmt.Printf("✓ Демон штатно остановлен (pid=%d)\n", pid)
		return 0

	case "restart":
		oldPID, st, err := ipc.RestartDaemon(cfg, cfgPath)
		if err != nil {
			fmt.Fprintf(os.Stderr, "✗ Ошибка перезапуска: %v\n", err)
			return 1
		}
		if oldPID > 0 {
			fmt.Printf("✓ Старый процесс остановлен (pid=%d)\n", oldPID)
		}
		fmt.Printf("✓ Демон перезапущен (новый pid=%d, режим=%s, память=%.1f МБ)\n",
			st.PID, st.PolicyMode, float64(st.RSSKB)/1024.0)
		return 0

	case "status":
		st := ipc.GetStatus(cfg)
		printStatusSummary(st)
		if !st.Running {
			return 3
		}
		return 0
	}
	return 2
}

func execConsoleTool(cfg config.Config, reader *bufio.Reader, fullLine, toolName string, rawArgs []string) {
	kv := make(map[string]string)
	for _, item := range rawArgs {
		idx := strings.IndexByte(item, '=')
		if idx <= 0 {
			toolName = fullLine
			kv = nil
			break
		}
		kv[strings.TrimSpace(item[:idx])] = strings.TrimSpace(item[idx+1:])
	}

	req := core.Request{
		Tool:   toolName,
		Args:   kv,
		Caller: "console",
	}

	timeout := time.Duration(cfg.Policy.CommandTimeoutSec+5) * time.Second
	resp, err := ipc.CallUnix(cfg.SocketPath, req, timeout)
	if err != nil {
		fmt.Fprintf(os.Stderr, "✗ Демон не запущен (%v). Введите '1' или 'start', чтобы запустить его.\n", err)
		return
	}

	if !resp.OK && resp.NeedsApproval {
		if promptConfirmationReader(reader, resp.Tool, string(resp.Risk), formatArgsInline(kv), resp.Error) {
			req.Approved = true
			resp, err = ipc.CallUnix(cfg.SocketPath, req, timeout)
			if err != nil {
				fmt.Fprintf(os.Stderr, "✗ Ошибка повторного запроса: %v\n", err)
				return
			}
		} else {
			fmt.Println("• Операция отменена.")
			return
		}
	}

	if !resp.OK {
		fmt.Fprintf(os.Stderr, "✗ Отказ политики [%s]: %s\n", resp.Tool, resp.Error)
		return
	}

	var pretty any
	if err := json.Unmarshal(resp.Data, &pretty); err != nil {
		fmt.Println(string(resp.Data))
		return
	}
	out, _ := json.MarshalIndent(pretty, "", "  ")
	fmt.Println(string(out))
}

func promptConfirmationReader(reader *bufio.Reader, toolName, risk, argsStr, reason string) bool {
	fmt.Printf("\n⚠️  Требуется подтверждение\n   Операция:  %s (риск: %s)\n", toolName, risk)
	if argsStr != "" {
		fmt.Printf("   Параметры: %s\n", argsStr)
	}
	fmt.Printf("   Причина:   %s\n   Выполнить? [y/N]: ", reason)

	line, err := reader.ReadString('\n')
	if err != nil && len(line) == 0 {
		return false
	}
	ans := strings.ToLower(strings.TrimSpace(line))
	return ans == "y" || ans == "yes" || ans == "д" || ans == "да"
}

func formatArgsInline(args map[string]string) string {
	if len(args) == 0 {
		return ""
	}
	parts := make([]string, 0, len(args))
	for k, v := range args {
		parts = append(parts, k+"="+v)
	}
	return strings.Join(parts, ", ")
}

func splitConsoleLine(input string) ([]string, error) {
	var tokens []string
	var buf strings.Builder
	var quote rune
	inToken := false

	for _, r := range input {
		switch {
		case quote != 0:
			if r == quote {
				quote = 0
			} else {
				buf.WriteRune(r)
			}
		case r == '"' || r == '\'':
			quote = r
			inToken = true
		case unicode.IsSpace(r):
			if inToken {
				tokens = append(tokens, buf.String())
				buf.Reset()
				inToken = false
			}
		default:
			buf.WriteRune(r)
			inToken = true
		}
	}
	if quote != 0 {
		return nil, io.ErrUnexpectedEOF
	}
	if inToken {
		tokens = append(tokens, buf.String())
	}
	return tokens, nil
}

func dispatchSmartInput(
	cfg *config.Config,
	cfgPath string,
	router *llm.Router,
	routerErr error,
	reader *bufio.Reader,
	fullLine, cmd string,
	rest []string,
) {
	if looksLikeRawDanger(fullLine) {
		execConsoleTool(*cfg, reader, fullLine, cmd, rest)
		return
	}

	specs, daemonUp := fetchDaemonSpecs(*cfg)
	if daemonUp && matchesRegisteredTool(specs, cmd, rest) {
		execConsoleTool(*cfg, reader, fullLine, cmd, rest)
		return
	}

	handleLLMChat(cfg, cfgPath, router, routerErr, reader, fullLine)
}

	specs, ok := fetchDaemonSpecs(*cfg)

					return "", err

			}
		}

		if !resp.OK {

		}

	}

func matchesRegisteredTool(specs []core.ToolSpec, cmd string, rest []string) bool {
	found := false
	for _, sp := range specs {
		short := sp.Name
		if idx := strings.IndexByte(short, '.'); idx >= 0 {
			short = short[idx+1:]
		}
		if cmd == sp.Name || cmd == short {
			found = true
			break
		}
	}
	if !found {
		return false
	}
	for _, arg := range rest {
		if strings.IndexByte(arg, '=') <= 0 {
			return false
		}
	}
	return true
}

func looksLikeRawDanger(line string) bool {
	low := strings.ToLower(strings.TrimSpace(line))
	prefixes := []string{
		"rm ", "mkfs", "dd if=", "chmod -r 777", ":(){", "shutdown", "reboot", "poweroff",
	}
	for _, p := range prefixes {
		if strings.HasPrefix(low, p) {
			return true
		}
	}
	return false
}

func handleServiceControl(action string, cfg config.Config, cfgPath string) int {
	switch action {
	case "start":
		st, already, err := ipc.StartDaemon(cfg, cfgPath)
		if err != nil {
			fmt.Fprintf(os.Stderr, "✗ Ошибка запуска: %v\n", err)
			return 1
		}
		if already {
			fmt.Printf("• Демон уже запущен (pid=%d, uptime=%s, память=%.1f МБ)\n",
				st.PID, st.Uptime, float64(st.RSSKB)/1024.0)
			return 0
		}
		fmt.Printf("✓ Демон запущен (pid=%d, режим=%s, память=%.1f МБ)\n",
			st.PID, st.PolicyMode, float64(st.RSSKB)/1024.0)
		return 0

	case "stop":
		pid, wasRunning, err := ipc.StopDaemon(cfg)
		if err != nil {
			fmt.Fprintf(os.Stderr, "✗ Ошибка остановки: %v\n", err)
			return 1
		}
		if !wasRunning {
			fmt.Println("• Демон уже остановлен.")
			return 0
		}
		fmt.Printf("✓ Демон штатно остановлен (pid=%d)\n", pid)
		return 0

	case "restart":
		oldPID, st, err := ipc.RestartDaemon(cfg, cfgPath)
		if err != nil {
			fmt.Fprintf(os.Stderr, "✗ Ошибка перезапуска: %v\n", err)
			return 1
		}
		if oldPID > 0 {
			fmt.Printf("✓ Старый процесс остановлен (pid=%d)\n", oldPID)
		}
		fmt.Printf("✓ Демон перезапущен (новый pid=%d, режим=%s, память=%.1f МБ)\n",
			st.PID, st.PolicyMode, float64(st.RSSKB)/1024.0)
		return 0

	case "status":
		st := ipc.GetStatus(cfg)
		printStatusSummary(st)
		if !st.Running {
			return 3
		}
		return 0
	}
	return 2
}

func execConsoleTool(cfg config.Config, reader *bufio.Reader, fullLine, toolName string, rawArgs []string) {
	kv := make(map[string]string)
	for _, item := range rawArgs {
		idx := strings.IndexByte(item, '=')
		if idx <= 0 {
			toolName = fullLine
			kv = nil
			break
		}
		kv[strings.TrimSpace(item[:idx])] = strings.TrimSpace(item[idx+1:])
	}

	req := core.Request{
		Tool:   toolName,
		Args:   kv,
		Caller: "console",
	}

	timeout := time.Duration(cfg.Policy.CommandTimeoutSec+5) * time.Second
	resp, err := ipc.CallUnix(cfg.SocketPath, req, timeout)
	if err != nil {
		fmt.Fprintf(os.Stderr, "✗ Демон не запущен (%v). Введите '1' или 'start', чтобы запустить его.\n", err)
		return
	}

	if !resp.OK && resp.NeedsApproval {
		if promptConfirmationReader(reader, resp.Tool, string(resp.Risk), formatArgsInline(kv), resp.Error) {
			req.Approved = true
			resp, err = ipc.CallUnix(cfg.SocketPath, req, timeout)
			if err != nil {
				fmt.Fprintf(os.Stderr, "✗ Ошибка повторного запроса: %v\n", err)
				return
			}
		} else {
			fmt.Println("• Операция отменена.")
			return
		}
	}

	if !resp.OK {
		fmt.Fprintf(os.Stderr, "✗ Отказ политики [%s]: %s\n", resp.Tool, resp.Error)
		return
	}

	var pretty any
	if err := json.Unmarshal(resp.Data, &pretty); err != nil {
		fmt.Println(string(resp.Data))
		return
	}
	out, _ := json.MarshalIndent(pretty, "", "  ")
	fmt.Println(string(out))
}

func promptConfirmationReader(reader *bufio.Reader, toolName, risk, argsStr, reason string) bool {
	fmt.Printf("\n⚠️  Требуется подтверждение\n   Операция:  %s (риск: %s)\n", toolName, risk)
	if argsStr != "" {
		fmt.Printf("   Параметры: %s\n", argsStr)
	}
	fmt.Printf("   Причина:   %s\n   Выполнить? [y/N]: ", reason)

	line, err := reader.ReadString('\n')
	if err != nil && len(line) == 0 {
		return false
	}
	ans := strings.ToLower(strings.TrimSpace(line))
	return ans == "y" || ans == "yes" || ans == "д" || ans == "да"
}

func formatArgsInline(args map[string]string) string {
	if len(args) == 0 {
		return ""
	}
	parts := make([]string, 0, len(args))
	for k, v := range args {
		parts = append(parts, k+"="+v)
	}
	return strings.Join(parts, ", ")
}

func splitConsoleLine(input string) ([]string, error) {
	var tokens []string
	var buf strings.Builder
	var quote rune
	inToken := false

	for _, r := range input {
		switch {
		case quote != 0:
			if r == quote {
				quote = 0
			} else {
				buf.WriteRune(r)
			}
		case r == '"' || r == '\'':
			quote = r
			inToken = true
		case unicode.IsSpace(r):
			if inToken {
				tokens = append(tokens, buf.String())
				buf.Reset()
				inToken = false
			}
		default:
			buf.WriteRune(r)
			inToken = true
		}
	}
	if quote != 0 {
		return nil, io.ErrUnexpectedEOF
	}
	if inToken {
		tokens = append(tokens, buf.String())
	}
	return tokens, nil
}
